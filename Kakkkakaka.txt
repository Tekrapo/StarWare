local roleCacheMod, roleCache, roleCacheStamp = nil, {}, 0
local function refreshRoleCache()
    local now = os.clock()
    if now - roleCacheStamp < 0.5 and next(roleCache) then return end
    roleCacheStamp = now
    if not roleCacheMod then
        local ok, m = pcall(function()
            return require(Replicated:WaitForChild("Modules"):WaitForChild("CurrentRoundClient"))
        end)
        if ok and type(m)=="table" then roleCacheMod = m end
    end
    local d = roleCacheMod and roleCacheMod.PlayerData
    if type(d) ~= "table" then return end
    local new = {}
    for name, info in pairs(d) do
        if type(info)=="table" and info.Role then new[name] = info.Role end
    end
    roleCache = new
end
local function scanTools(plr)
    local bag, char = plr:FindFirstChild("Backpack"), plr.Character
    local k, g = false, false
    local function scan(c)
        if not c then return end
        for _, t in ipairs(c:GetChildren()) do
            if t:IsA("Tool") then
                local n = t.Name:lower()
                if n:find("knife") then k = true end
                if n:find("gun") or n:find("revolver") or n:find("pistol") then g = true end
            end
        end
    end
    scan(bag); scan(char)
    if k then return "m" end
    if g then return "s" end
    return nil
end
local function cachedRole(plr)
    refreshRoleCache()
    local rc = roleCache[plr.Name]
    if rc == "Murderer" then return "m" end
    if rc == "Sheriff" or rc == "Hero" then return "s" end
    return scanTools(plr)
end
local function alive(p)
    local c = p.Character; if not c then return false end
    local h = c:FindFirstChildOfClass("Humanoid")
    return h and h.Health > 0
end
local function nearestPlayer()
    local best, bd = nil, math.huge
    local myp = HRP(); if not myp then return nil end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                local d = (hrp.Position - myp.Position).Magnitude
                if d < bd then bd = d; best = p end
            end
        end
    end
    return best
end

local target_p, target_char, target_part
local function refreshTarget()
    refreshRoleCache()
    local best, bd = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and alive(p) and cachedRole(p) == "m" then
            local h = p.Character and p.Character:FindFirstChild("Head")
            if h then
                local d = (h.Position - Cam.CFrame.Position).Magnitude
                if d < bd then bd = d; best = p end
            end
        end
    end
    if best ~= target_p then target_p = best; target_char, target_part = nil, nil end
    if not best then return end
    local c = best.Character
    if c ~= target_char then target_char, target_part = nil, nil end
    if not target_part or not target_part.Parent then
        target_part = c:FindFirstChild("Head") or c:FindFirstChild("HumanoidRootPart")
            or c:FindFirstChild("UpperTorso") or c:FindFirstChild("Torso")
    end
end

local track = { part=nil, pos=nil, t=0, v=Vector3.zero, ready=false }
local function trackStep(now)
    local p = target_part
    if not p or not p.Parent then return end
    local pos = p.Position
    if p ~= track.part or not track.pos then
        track.part, track.pos, track.t = p, pos, now
        track.v, track.ready = Vector3.zero, false
        return
    end
    local dt = now - track.t
    if dt <= 0 or dt > 0.4 then
        track.part, track.pos, track.t = p, pos, now
        return
    end
    local inst = (pos - track.pos) / dt
    track.v = track.v * 0.7 + inst * 0.3
    track.pos, track.t = pos, now
    if track.v.Magnitude > 0.5 then track.ready = true end
end

-- //////////// AIM AUTOTUNE ////////////
local function tuneAim()
    if not S.silent then return end
    local ok, ms = pcall(function()
        return Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
    end)
    local rtt = (ok and type(ms)=="number" and ms > 0) and (ms/1000) or 0.1
    rtt = math.clamp(rtt, 0.02, 0.6)
    local log = AIM.sampleLog
    log[#log+1] = rtt
    if #log > 10 then table.remove(log, 1) end
    local sum = 0
    for _, v in ipairs(log) do sum = sum + v end
    local mean = sum / #log
    local jit = 0
    for _, v in ipairs(log) do jit = jit + math.abs(v - mean) end
    jit = #log > 1 and (jit / #log) or 0
    AIM.ping, AIM.jitter = mean, jit
    local span = math.clamp(mean + jit * 0.6, 0.03, 0.45)
    AIM.antiJump = false
    local tc = target_char
    if tc then
        local h = tc:FindFirstChildOfClass("Humanoid")
        if h then
            local st = h:GetState()
            local air = st == Enum.HumanoidStateType.Jumping
                or st == Enum.HumanoidStateType.Freefall
                or h.FloorMaterial == Enum.Material.Air
            AIM.airborne = air
            if air then span = span * 1.45; AIM.antiJump = true end
        end
    end
    AIM.span = math.clamp(span, 0.03, 0.6)
    AIM.smooth = math.clamp(0.9 / (1 + AIM.span * 8), 0.05, 0.5)
    local status = string.format("ping %.0fms | jit %.0fms | span %.2f | sm %.2f%s",
        AIM.ping*1000, AIM.jitter*1000, AIM.span, AIM.smooth,
        AIM.antiJump and " | AIR" or "")
    if status ~= AIM.lastStatus then
        AIM.lastStatus = status
        if getgenv().ANON_STATUS_TXT then pcall(getgenv().ANON_STATUS_TXT, status) end
    end
end

local HITS = { "Head","HumanoidRootPart","UpperTorso","Torso",
    "LeftUpperArm","RightUpperArm","LeftUpperLeg","RightUpperLeg",
    "LeftHand","RightHand","LeftFoot","RightFoot" }
local function aimPoint()
    if not target_char then return nil end
    local span = AIM.span
    local g = workspace.Gravity
    for _, n in ipairs(HITS) do
        local p = target_char:FindFirstChild(n)
        if p and p:IsA("BasePart") then
            if track.ready and span > 0.01 then
                local pos = p.Position
                return Vector3.new(
                    pos.X + track.v.X * span,
                    pos.Y + track.v.Y * span - 0.5 * g * span * span,
                    pos.Z + track.v.Z * span
                )
            end
            return p.Position
        end
    end
end

local ws, orig_mouse, orig_screen
local function getWS()
    if ws then return ws end
    local ok, m = pcall(function()
        return require(Replicated:WaitForChild("ClientServices"):WaitForChild("WeaponService"))
    end)
    if ok and type(m)=="table" then ws = m end
    return ws
end
local function resolveAim()
    if not S.silent then return nil end
    if not target_part or not target_part.Parent then pcall(refreshTarget) end
    if not target_part or not target_part.Parent then
        local p = aimPoint(); if not p then return nil end
        return CFrame.new(p)
    end
    local p = aimPoint() or target_part.Position
    return CFrame.new(p)
end
local function installHooks()
    local m = getWS(); if not m then return end
    pcall(function() setreadonly(m, false) end)
    if type(m.GetMouseTargetCFrame) == "function" and not orig_mouse then
        orig_mouse = m.GetMouseTargetCFrame
        m.GetMouseTargetCFrame = function(self, ...)
            local cf = resolveAim()
            if cf then return cf end
            return orig_mouse(self, ...)
        end
    end
    if type(m.GetTargetPosition) == "function" and not orig_screen then
        orig_screen = m.GetTargetPosition
        m.GetTargetPosition = function(self, x, y, ...)
            local cf = resolveAim()
            if cf then return cf end
            return orig_screen(self, x, y, ...)
        end
    end
    if hookmetamethod and getnamecallmethod and not ANON.ncHooked then
        ANON.ncHooked = true
        pcall(function()
            local nc = newcclosure or function(f) return f end
            local old
            old = hookmetamethod(game, "__namecall", nc(function(self, ...)
                local method
                local okm, mm = pcall(getnamecallmethod)
                if okm then method = mm end
                if (method == "Raycast" or method == "FindPartOnRay") and S.silent and self == workspace then
                    local origin, direction
                    if method == "Raycast" then origin, direction = ...
                    else
                        local ray = ...
                        if typeof(ray) == "Ray" then origin, direction = ray.Origin, ray.Direction end
                    end
                    if origin and direction and typeof(origin) == "Vector3" and typeof(direction) == "Vector3" and direction.Magnitude > 30 then
                        local okw, hit = pcall(function()
                            local h0 = HRP()
                            if not (h0 and target_char and target_part and target_part.Parent) then return nil end
                            if (origin - h0.Position).Magnitude > 25 then return nil end
                            local p = aimPoint() or target_part.Position
                            local dir = p - origin
                            if dir.Magnitude <= 1 then return nil end
                            local wp = RaycastParams.new()
                            wp.FilterType = Enum.RaycastFilterType.Include
                            wp.FilterDescendantsInstances = { target_char }
                            wp.IgnoreWater = true
                            local r = old(self, origin, dir.Unit * 5000, wp)
                            if r and r.Instance then return r end
                        end)
                        if okw and hit then
                            if method == "Raycast" then return hit end
                            return hit.Instance, hit.Position, hit.Normal, hit.Material
                        end
                    end
                end
                return old(self, ...)
            end))
        end)
    end
end

local function isProjectile(p)
    if not p or not p:IsA("BasePart") then return false end
    local n = p.Name:lower()
    return n:find("bullet") or n:find("knife") or n:find("blade")
        or n:find("arrow") or n:find("projectile") or n:find("thrown")
end
local function redirectProj(p)
    if not S.capture then return end
    local tgt = target_part
    if not tgt or not tgt.Parent then return end
    p.CFrame = CFrame.new(p.Position, tgt.Position)
    local v = p.AssemblyLinearVelocity
    if v and v.Magnitude > 0 then
        p.AssemblyLinearVelocity = (tgt.Position - p.Position).Unit * v.Magnitude
    end
end
workspace.DescendantAdded:Connect(function(o)
    if S.capture and isProjectile(o) then
        redirectProj(o); task.wait(0.01); redirectProj(o)
    end
end)

local function to_pos(v)
    if typeof(v)=="Vector3" then return v end
    if typeof(v)=="CFrame" then return v.Position end
    if typeof(v)=="Instance" then
        if v:IsA("Attachment") then return v.WorldPosition end
        if v:IsA("BasePart") then return v.Position end
    end
    return nil
end
local function drawTracer(a, b)
    local sp = Instance.new("Part")
    sp.Transparency=1; sp.Anchored=true; sp.CanCollide=false; sp.CanQuery=false
    sp.Size=Vector3.new(1,1,1); sp.CFrame=CFrame.new(a); sp.Parent=workspace
    Instance.new("Attachment", sp)
    local ep = Instance.new("Part")
    ep.Transparency=1; ep.Anchored=true; ep.CanCollide=false; ep.CanQuery=false
    ep.Size=Vector3.new(1,1,1); ep.CFrame=CFrame.new(b); ep.Parent=workspace
    Instance.new("Attachment", ep)
    local beam = Instance.new("Beam")
    beam.FaceCamera=true; beam.Width0=0.25; beam.Width1=0.25
    beam.LightEmission=3; beam.LightInfluence=0; beam.Brightness=2.5
    beam.Color=ColorSequence.new(FC.tracer)
    beam.Transparency=NumberSequence.new(0.1)
    beam.Attachment0=sp.Attachment; beam.Attachment1=ep.Attachment
    beam.Parent=sp
    Debris:AddItem(sp, 1); Debris:AddItem(ep, 1)
    task.delay(0.8, function() if beam.Parent then
        Tween:Create(beam, TweenInfo.new(0.2), {Width0=0, Width1=0}):Play()
    end end)
end
local tracerConn
local function enableTracer()
    if tracerConn then return end
    local m = getWS()
    local ev = m and m.GunFired
    if typeof(ev) ~= "Instance" then return end
    tracerConn = ev.OnClientEvent:Connect(function(gun, a, b)
        if not S.gunTracer then return end
        local c = LP.Character; if not c then return end
        if typeof(gun)~="Instance" or not gun:IsDescendantOf(c) then return end
        local pa, pb = to_pos(a), to_pos(b)
        if pa and pb then task.spawn(drawTracer, pa, pb) end
    end)
end
local function disableTracer()
    if tracerConn then tracerConn:Disconnect(); tracerConn=nil end
end

-- //////////// KILL AURA ////////////
local function killAuraTick()
    if not S.killAura then return end
    local c = LP.Character; if not c then return end
    local knife = c:FindFirstChild("Knife")
    if not knife then
        local bp = LP:FindFirstChildOfClass("Backpack")
        knife = bp and bp:FindFirstChild("Knife")
        if knife then
            local h = c:FindFirstChildOfClass("Humanoid")
            if h then pcall(function() h:EquipTool(knife) end) end
        end
        return
    end
    local myp = HRP(); if not myp then return end
    local ev = knife:FindFirstChild("Events")
    local stabbed = ev and ev:FindFirstChild("KnifeStabbed")
    local touched = ev and ev:FindFirstChild("HandleTouched")
    if not stabbed or not touched then return end
    local fired = false
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and alive(plr) then
            local tc = plr.Character
            local tp = tc and (tc:FindFirstChild("HumanoidRootPart") or tc:FindFirstChild("Head"))
            if tp and (tp.Position - myp.Position).Magnitude <= S.auraDist then
                if not fired then pcall(function() stabbed:FireServer() end); fired=true end
                pcall(function() touched:FireServer(tp) end)
            end
        end
    end
end

local function startFly()
    if flyConn then flyConn:Disconnect(); flyConn = nil end
    local c = LP.Character; if not c then return end
    local hrp = c:FindFirstChild("HumanoidRootPart")
    local h = c:FindFirstChildOfClass("Humanoid")
    if not hrp or not h then return end
    h.PlatformStand = true
    if IS_MOBILE then startFlyStick() end
    flyConn = RS.RenderStepped:Connect(function(dt)
        if not S.fly or not ANON.loaded then return end
        local c2 = LP.Character; if not c2 then return end
        local hrp2 = c2:FindFirstChild("HumanoidRootPart")
        local h2 = c2:FindFirstChildOfClass("Humanoid")
        if not hrp2 or not h2 then return end
        pcall(function() h2.PlatformStand = true end)
        local cam = workspace.CurrentCamera
        local look = cam.CFrame.LookVector
        local right = cam.CFrame.RightVector
        local fl = Vector3.new(look.X, 0, look.Z); if fl.Magnitude > 0 then fl = fl.Unit end
        local fr = Vector3.new(right.X, 0, right.Z); if fr.Magnitude > 0 then fr = fr.Unit end
        local dir = Vector3.zero
        if UIS.KeyboardEnabled then
            if UIS:IsKeyDown(Enum.KeyCode.W) then dir = dir + look end
            if UIS:IsKeyDown(Enum.KeyCode.S) then dir = dir - look end
            if UIS:IsKeyDown(Enum.KeyCode.A) then dir = dir - right end
            if UIS:IsKeyDown(Enum.KeyCode.D) then dir = dir + right end
            if UIS:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0,1,0) end
            if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.new(0,1,0) end
        end
        if IS_MOBILE then
            if flyStick.pos.Magnitude > 0.05 then
                dir = dir + fl * flyStick.pos.Y + fr * flyStick.pos.X
            end
            if flyStick.up then dir = dir + Vector3.new(0,1,0) end
            if flyStick.dn then dir = dir - Vector3.new(0,1,0) end
        end
        if dir.Magnitude > 0 then dir = dir.Unit end
        local step = math.min(dt or 0.016, 0.05)
        if dir.Magnitude > 0.1 then
            hrp2.CFrame = hrp2.CFrame + dir * S.flySpeed * step
        end
        pcall(function()
            hrp2.AssemblyLinearVelocity = Vector3.zero
            hrp2.AssemblyAngularVelocity = Vector3.zero
        end)
    end)
end

    destroyFlyStick()

-- //////////// BACKTRACK ////////////
local BT = { model=nil, cache={}, hist={}, cap=128, first=1, n=0, ping=0.15, lastping=0 }
local function btDestroy()
    if BT.model then pcall(function() BT.model:Destroy() end); BT.model=nil end
    BT.cache = {}; BT.hist = {}; BT.first=1; BT.n=0
end
local function btBuild()
    btDestroy()
    local char = LP.Character; if not char then return end
    local rhrp = char:FindFirstChild("HumanoidRootPart"); if not rhrp then return end
    char.Archivable = true
    local ok, m = pcall(function() return char:Clone() end)
    char.Archivable = false
    if not ok or not m then return end
    local rparts = {}
    for _, o in ipairs(char:GetDescendants()) do
        if o:IsA("BasePart") then rparts[#rparts+1] = o end
    end
    local ci = 0
    for _, o in ipairs(m:GetDescendants()) do
        if o:IsA("BasePart") then
            o.Anchored=true; o.CanCollide=false; o.CanQuery=false; o.CanTouch=false
            o.Massless=true; o.CastShadow=false
            if o.Name=="HumanoidRootPart" then o.Transparency=1
            else o.Material=Enum.Material.ForceField; o.Color=Color3.fromRGB(170,100,255); o.Transparency=0 end
            ci = ci + 1
            BT.cache[#BT.cache+1] = {o, rparts[ci]}
        elseif o:IsA("Script") or o:IsA("LocalScript") or o:IsA("ModuleScript")
            or o:IsA("ParticleEmitter") or o:IsA("Beam") or o:IsA("Trail")
            or o:IsA("Sound") or o:IsA("Decal") or o:IsA("Texture") then
            pcall(function() o:Destroy() end)
        end
    end
    local hh = m:FindFirstChildOfClass("Humanoid"); if hh then hh:Destroy() end
    m.Name = "\0"; m.Parent = workspace
    BT.model = m
end
local function btUpdate()
    if not S.backtrack then return end
    local char = LP.Character; if not char then return end
    local rhrp = char:FindFirstChild("HumanoidRootPart"); if not rhrp then return end
    if not BT.model or not BT.model.Parent then btBuild(); if not BT.model then return end end
    local now = os.clock()
    local base_cf = rhrp.CFrame
    if BT.n < BT.cap then BT.n = BT.n + 1
    else BT.first = BT.first % BT.cap + 1 end
    local idx = (BT.first + BT.n - 2) % BT.cap + 1
    local slot = BT.hist[idx]; if not slot then slot = {}; BT.hist[idx] = slot end
    slot[1] = now; slot[2] = base_cf
    if now - BT.lastping >= 0.2 then
        BT.lastping = now
        local ok, val = pcall(function()
            return Stats.Network.ServerStatsItem["Data Ping"]:GetValue()/1000
        end)
        BT.ping = math.clamp((ok and val) or 0.15, 0.05, 0.6)
    end
    local tgt = now - BT.ping
    local cf = base_cf
    for k = BT.n, 1, -1 do
        local s = BT.hist[(BT.first + k - 2) % BT.cap + 1]
        if s and s[1] and s[1] <= tgt then cf = s[2]; break end
    end
    while BT.n > 0 and BT.hist[BT.first] and BT.hist[BT.first][1] < now - 1 do
        BT.first = BT.first % BT.cap + 1; BT.n = BT.n - 1
    end
    local inv = rhrp.CFrame:Inverse()
    for i = 1, #BT.cache do
        local cp, rpv = BT.cache[i][1], BT.cache[i][2]
        if cp and cp.Parent and rpv and rpv.Parent then
            cp.CFrame = cf * (inv * rpv.CFrame)
        end
    end
end

                if k>=1 or not p.Parent then cn:Disconnect(); if p.Parent then p:Destroy() end; return end
                local sz = s*(1-k*0.6)
                p.Size=Vector3.new(0.15,sz,sz)
                p.Transparency=0.15+k*0.85
            end)
        end
    end)
end

end

-- //////////// TROLL ////////////
local TROLL = { flinging=false, sitSpam=false, follow=false, followTarget=nil, sitLast=0 }
local function flingCore(thrp, thum, time)
    local c = LP.Character
    local hrp = c and c:FindFirstChild("HumanoidRootPart"); if not hrp then return end
    local h2 = c and c:FindFirstChildOfClass("Humanoid"); if not h2 then return end
    if thum and thum.Sit then return end
    local orig = hrp.CFrame
    local cam = workspace.CurrentCamera
    local oldSubj = cam.CameraSubject
    local oldFall = workspace.FallenPartsDestroyHeight
    pcall(function() workspace.FallenPartsDestroyHeight = 0/0 end)
    pcall(function() cam.CameraSubject = thrp end)
    local se = h2:GetStateEnabled(Enum.HumanoidStateType.Seated)
    pcall(function() h2:SetStateEnabled(Enum.HumanoidStateType.Seated, false) end)
    local t0 = os.clock()
    local dur = time or 1.6
    local ang = 0
    while os.clock() - t0 < dur and hrp.Parent and thrp.Parent do
        local tv = thrp.Velocity
        if tv.Magnitude > 50 and thum then tv = thum.MoveDirection * thum.WalkSpeed end
        local lead = thrp.Position + tv * 0.08
        ang = ang + 110
        local r = ang % 360
        local off
        if r < 90 then off = Vector3.new(0,1.5,0)
        elseif r < 180 then off = Vector3.new(0,-1.5,0)
        elseif r < 270 then off = Vector3.new(1.5,0,0)
        else off = Vector3.new(-1.5,0,0) end
        local cf = CFrame.new(lead + off) * CFrame.Angles(math.rad(ang),0,0)
        pcall(function()
            hrp.CFrame = cf
            c:SetPrimaryPartCFrame(cf)
            hrp.Velocity = Vector3.new(9e7,9e7*10,9e7)
            hrp.RotVelocity = Vector3.new(9e8,9e8,9e8)
            hrp.AssemblyLinearVelocity = Vector3.new(9e7,9e7*10,9e7)
            hrp.AssemblyAngularVelocity = Vector3.new(9e8,9e8,9e8)
        end)
        RS.Heartbeat:Wait()
    end
    pcall(function() h2:SetStateEnabled(Enum.HumanoidStateType.Seated, se) end)
    if hrp.Parent then
        pcall(function()
            hrp.CFrame = orig
            h2:ChangeState(Enum.HumanoidStateType.GettingUp)
            hrp.Velocity = Vector3.zero
            hrp.RotVelocity = Vector3.zero
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end)
    end
    pcall(function() workspace.FallenPartsDestroyHeight = oldFall end)
    if oldSubj then pcall(function() cam.CameraSubject = oldSubj end) end
end
local function flingTarget(target)
    if TROLL.flinging then return end
    local tchar = target and target.Character; if not tchar then return end
    local thrp = tchar:FindFirstChild("HumanoidRootPart") or tchar:FindFirstChild("Head"); if not thrp then return end
    local thum = tchar:FindFirstChildOfClass("Humanoid")
    TROLL.flinging = true
    task.spawn(function() flingCore(thrp, thum, 1.6); TROLL.flinging = false end)
end
local function flingNearest()
    local t = nearestPlayer(); if t then flingTarget(t) end
end
local function flingAll()
    if TROLL.flinging then return end
    TROLL.flinging = true
    task.spawn(function()
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LP and p.Character then
                local thrp = p.Character:FindFirstChild("HumanoidRootPart") or p.Character:FindFirstChild("Head")
                local thum = p.Character:FindFirstChildOfClass("Humanoid")
                if thrp then flingCore(thrp, thum, 1.1); task.wait(0.15) end
            end
        end
        TROLL.flinging = false
    end)
end
local function stopTroll()
    TROLL.sitSpam=false; TROLL.follow=false; TROLL.followTarget=nil; TROLL.flinging=false
end
local function trollTick()
    if TROLL.flinging then return end
    if TROLL.sitSpam then
        local now = os.clock()
        if now - TROLL.sitLast > 0.1 then
            TROLL.sitLast = now
            local h = hum(); if h then h.Sit = not h.Sit end
        end
    end
    if TROLL.follow and TROLL.followTarget then
        local hrp = HRP()
        local thrp = TROLL.followTarget.Character and TROLL.followTarget.Character:FindFirstChild("HumanoidRootPart")
        if hrp and thrp then
            hrp.CFrame = CFrame.new(thrp.Position + thrp.CFrame.LookVector * -3)
        end
    end
end

            roleAcc = now
            refreshTarget()
        end
        trackStep(now)
        killAuraTick()
        trollTick()
        btUpdate()
        if now - antAcc >= 0.2 then
            antAcc = now
            local hrp = HRP()
            if hrp then
                if hrp.Position.Y > workspace.FallenPartsDestroyHeight + 30 then
                    lastSafe = hrp.CFrame
                elseif S.antiFall and lastSafe then
                    hrp.CFrame = lastSafe; hrp.Velocity = Vector3.zero
                end
            end
        end
        if now - espAcc >= 0.4 and S.esp then
            espAcc = now
            pcall(applyESP)
        end
    end)
end)

    ANON.loaded = false
    stopFly(); stopNoclip(); stopInfJump()
    stopTroll()

    if ANON.hb then pcall(function() ANON.hb:Disconnect() end) ANON.hb = nil end

    if m then
        pcall(function() setreadonly(m, false) end)
        if orig_mouse then pcall(function() m.GetMouseTargetCFrame = orig_mouse end) end
        if orig_screen then pcall(function() m.GetTargetPosition = orig_screen end) end
    end
    pcall(function() Fluent:Destroy() end)
    for _, fn in ipairs({clTrail, clParts, clHat, clFoot, clRB, clGlow, clSnow, clLight,
                         clOrb, clGhost, clSky, clBo, clAtmo, clFB, clSkybox, clBloom,
                         disableTracer}) do pcall(fn) end
    pcall(function() if S.fog then setFog(false) end end)
    local pg = LP:FindFirstChild("PlayerGui")
    if pg then
        local f = pg:FindFirstChild("ANON_FLY_STICK"); if f then f:Destroy() end
    end
    local c = LP.Character
    if c then
        for _, x in ipairs(c:GetDescendants()) do
            if x.Name == "anon_esp" then x:Destroy() end
        end
    end
    for _, x in ipairs(workspace:GetChildren()) do
        if x.Name == "anon_esp" then x:Destroy() end
    end
    getgenv().ANON = nil
    getgenv().ANON_STATUS_TXT = nil
    print("[ANON] unloaded")
end
