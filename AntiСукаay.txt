local roleCacheMod = nil
local roleCache = {}
local roleCacheStamp = 0
local function refreshRoleCache()
    local now = os.clock()
    if now - roleCacheStamp < 0.4 and next(roleCache) then return end
    roleCacheStamp = now
    if not roleCacheMod then
        local ok, m = pcall(function()
            return require(Replicated:WaitForChild("Modules"):WaitForChild("CurrentRoundClient"))
        end)
        if ok and type(m) == "table" then roleCacheMod = m end
    end
    local d = roleCacheMod and roleCacheMod.PlayerData
    if type(d) ~= "table" then return end
    local new = {}
    for name, info in pairs(d) do
        if type(info) == "table" and info.Role then new[name] = info.Role end
    end
    roleCache = new
end

local EC = { rtt=0, jitter=0, seen=false }
local EC_samples = {}

local function push_rtt(v)
    table.insert(EC_samples, v)
    if #EC_samples > 12 then table.remove(EC_samples, 1) end
end
local function rtt_p90()
    if #EC_samples == 0 then return EC.rtt end
    local s = table.clone(EC_samples)
    table.sort(s)
    return s[math.max(1, math.floor(#s * 0.9))]
end

local function SA_rayBlocked(a, b, ignore)
    local list = { workspace.CurrentCamera }
    for _, x in ipairs(ignore or {}) do table.insert(list, x) end
    local p = RaycastParams.new()
    p.FilterType = Enum.RaycastFilterType.Exclude
    for _ = 1, 8 do
        p.FilterDescendantsInstances = list
        local hit = workspace:Raycast(a, b - a, p)
        if not hit then return false end
        local i = hit.Instance
        if i.Transparency >= 0.9 or not i.CanCollide then table.insert(list, i)
        else return true end
    end
    return true
end

local function SA_anyVisible(targetChar, origin)
    for _, n in ipairs(SA_BODY) do
        local pt = targetChar:FindFirstChild(n)
        if pt and pt:IsA("BasePart") then
            if not SA_rayBlocked(origin, pt.Position, { targetChar }) then return true end
        end
    end

end

local function SA_targetScore(p, camPos, camLook)
    local c = p.Character
    if not c then return math.huge end
    local head = c:FindFirstChild("Head")
    local hrp  = c:FindFirstChild("HumanoidRootPart")
    local h    = c:FindFirstChildOfClass("Humanoid")
    if not head or not hrp or not h then return math.huge end
    local toHead = head.Position - camPos
    local dist = toHead.Magnitude
    if dist < 0.5 then dist = 0.5 end
    local angle = math.deg(math.acos(math.clamp(camLook:Dot(toHead.Unit), -1, 1)))
    if angle > SA.fov * 0.5 then return math.huge end
    local jumpMul = 1
    if SA.antiJump and h.FloorMaterial == Enum.Material.Air then
        local vy = hrp.AssemblyLinearVelocity.Y
        if vy > 8 then jumpMul = 1.6 elseif vy > 0 then jumpMul = 1.1
        elseif vy > -25 then jumpMul = 0.85 else jumpMul = 0.6 end
    end
    local occluded = SA_rayBlocked(camPos, head.Position, { c })
    local anglePenalty = 1 + (angle / math.max(SA.fov, 1)) * SA.angleBias * 4
    local score = dist * anglePenalty * jumpMul
    if occluded then score = score * 1.4 end
    return score
end

local function refresh_target()
    refreshRoleCache()
    if not Cam then Cam = workspace.CurrentCamera end
    if not Cam then return end
    local camPos, camLook = Cam.CFrame.Position, Cam.CFrame.LookVector
    if target_p and target_p.Character then
        local h = target_p.Character:FindFirstChild("Humanoid")
        if h and h.Health > 0 and cachedRole(target_p) == "m" then
            local s = SA_targetScore(target_p, camPos, camLook)
            if s < math.huge then
                local bs = s / 1.4
                local better
                for _, p in ipairs(Players:GetPlayers()) do
                    if p ~= LP and p ~= target_p and alive(p) and cachedRole(p) == "m" then
                        local ps = SA_targetScore(p, camPos, camLook)
                        if ps < bs then bs = ps; better = p end
                    end
                end
                if not better then return end
            end
        end
    end
    local best, bs = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and alive(p) and cachedRole(p) == "m" then
            local s = SA_targetScore(p, camPos, camLook)
            if s < bs then bs = s; best = p end
        end
    end
    if best ~= target_p then
        if SA_lock.lastChar ~= best then SA_lock.lastChar = best; SA_lock.blendT = 1.0 end
        target_p = best; target_char, target_part, target_hum = nil, nil, nil
    end
    if not best then return end
    local c = best.Character
    if c ~= target_char then target_char = c; target_part, target_hum = nil, nil end
    if not c then return end
    if not target_part or not target_part.Parent then
        target_part = c:FindFirstChild("HumanoidRootPart")
            or c:FindFirstChild("UpperTorso") or c:FindFirstChild("Torso")
    end
    if not target_hum or not target_hum.Parent then
        target_hum = c:FindFirstChildOfClass("Humanoid")
    end
end

end

local function SA_getVel(root)
    local list = SA_hist[root]
    if not list or #list < 2 then return root.AssemblyLinearVelocity end
    local vx, vy, vz, w = 0, 0, 0, 0
    for i = 2, #list do
        local a, b = list[i-1], list[i]
        local dt = b[1] - a[1]
        if dt > 0.008 and dt < 0.2 then
            vx = vx + ((b[2].X - a[2].X) / dt) * dt
            vy = vy + ((b[2].Y - a[2].Y) / dt) * dt
            vz = vz + ((b[2].Z - a[2].Z) / dt) * dt
            w = w + dt
        end
    end
    if w <= 0 then return root.AssemblyLinearVelocity end
    local v = Vector3.new(vx/w, vy/w, vz/w)
    if v.Magnitude > 200 then v = v.Unit * 200 end
    return v
end

local function SA_jumpScore(hum2, root)
    if not hum2 or not root then return 1 end
    local st = hum2:GetState()
    if st ~= Enum.HumanoidStateType.Jumping and st ~= Enum.HumanoidStateType.Freefall then return 1.0 end
    local vy = root.AssemblyLinearVelocity.Y
    if vy > 8 then return 0.35 elseif vy > 0 then return 0.6
    elseif vy > -25 then return 0.9 else return 1.0 end
end

local function SA_predict(hum2, root, lead, aimHint)
    if not SA.autoPred then
        local v = SA_getVel(root)
        return Vector3.new(v.X * lead, math.clamp(v.Y * lead, -3, 3), v.Z * lead)
    end
    local physV = root.AssemblyLinearVelocity
    local md = hum2 and hum2.MoveDirection or Vector3.zero
    local vBase
    if hum2 and hum2.FloorMaterial == Enum.Material.Air then
        local dy = physV.Y * lead - 0.5 * workspace.Gravity * lead * lead
        return Vector3.new(physV.X * lead, dy, physV.Z * lead)
    end
    local horiz = Vector3.new(physV.X, 0, physV.Z)
    if horiz.Magnitude > 1 then vBase = horiz
    elseif md.Magnitude > 0.1 and hum2 then vBase = md.Unit * hum2.WalkSpeed
    else vBase = SA_getVel(root) end
    if aimHint then
        local toAim = aimHint - root.Position
        if toAim.Magnitude > 0.01 then
            local fwd = toAim.Unit
            local right = fwd:Cross(Vector3.new(0, 1, 0))
            if right.Magnitude > 0.01 then
                right = right.Unit
                local vR = vBase:Dot(right)
                local vF = vBase:Dot(fwd)
                vBase = fwd * vF + right * (vR * SA.lateralBias)
            end
        end
    end
    return Vector3.new(vBase.X * lead, math.clamp(vBase.Y * lead, -3, 3), vBase.Z * lead)
end

local function SA_confidence(c, hum2, root, muzzle)
    if not SA.bestMoment then return 1 end
    local s = 1
    if SA.antiJump then s = s * SA_jumpScore(hum2, root) end
    local wvis, wtot = 0, 0
    for _, n in ipairs(SA_BODY) do
        local pt = c:FindFirstChild(n)
        if pt and pt:IsA("BasePart") then
            local w = BODY_WEIGHT[n] or 1
            wtot = wtot + w
            if not SA_rayBlocked(muzzle, pt.Position, { c }) then wvis = wvis + w end
        end
    end
    if wtot > 0 then s = s * (0.35 + 0.65 * (wvis / wtot)) end
    if EC.jitter > 0.05 then s = s * math.clamp(1 - (EC.jitter - 0.05) * 3, 0.5, 1) end
    local phase = SA_phase[c]
    if hum2 and hum2.MoveDirection.Magnitude > 0.1 then
        if phase and phase.lastMd and phase.lastMd.Magnitude > 0.1 then
            local dot = hum2.MoveDirection.Unit:Dot(phase.lastMd.Unit)
            if dot < 0.5 then s = s * 0.7 end
        end
        SA_phase[c] = { lastMd = hum2.MoveDirection }
    else SA_phase[c] = nil end
    return s
end

local function SA_findWbSpot(mRoot, muzzle)
    local mChar = mRoot.Parent
    local toM = muzzle - mRoot.Position
    local base = math.atan2(toM.Z, toM.X)
    for _, dist in ipairs({ 1.5, 2.5, 3.5, 5, 7, 9 }) do
        for k = 0, 23 do
            local ang = base + (k % 2 == 0 and 1 or -1) * math.ceil(k / 2) * (math.pi / 12)
            for _, h in ipairs({ 1, 2, 3, 4.5, 6 }) do
                local p = mRoot.Position + Vector3.new(math.cos(ang)*dist, h, math.sin(ang)*dist)
                if not SA_rayBlocked(p, mRoot.Position, { mChar }) and SA_anyVisible(mChar, p) then return p end
            end
        end
    end
end

local function SA_findQuietSpot(mRoot, myChar)
    local root = myChar:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local facing = CFrame.lookAt(root.Position, Vector3.new(mRoot.Position.X, root.Position.Y, mRoot.Position.Z))
    for _, off in ipairs({
        Vector3.new(0, 0, 4), Vector3.new(0, 0, -4),
        Vector3.new(4, 0, 0), Vector3.new(-4, 0, 0),
        Vector3.new(0, 5, 3), Vector3.new(0, 2, 2),
    }) do
        local pos = (facing * CFrame.new(off)).Position
        if not SA_rayBlocked(root.Position, pos) and not SA_rayBlocked(pos, mRoot.Position) then
            return CFrame.lookAt(pos, mRoot.Position)
        end
    end
end

resolve = function()
    if not SA.enabled then return nil end
    if not target_p or not target_p.Character then
        pcall(refresh_target)
        if not target_p then return nil end
    end
    local c = target_char
    if not c or not c.Parent then pcall(refresh_target); c = target_char end
    if not c then return nil end
    local root = c:FindFirstChild("HumanoidRootPart")
    if not root then return nil end
    local hum2 = c:FindFirstChildOfClass("Humanoid")
    local lead = math.clamp(rtt_p90() * 0.5 + EC.jitter * 0.5 + 0.04, 0.05, 0.32)
    local off = SA_predict(hum2, root, lead)
    local myChar = LP.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    local att = myRoot and myRoot:FindFirstChild("GunRaycastAttachment")
    local muzzle = att and att.WorldPosition or (myRoot and myRoot.Position)
    if not muzzle then return nil end
    local hint = root.Position + off
    local off2 = SA_predict(hum2, root, lead, hint)
    off = off * 0.5 + off2 * 0.5
    local conf = SA_confidence(c, hum2, root, muzzle)
    if conf < SA.confFloor then return nil end
    local wbSpot
    if SA.wb and SA_wbPush == nil and SA_rayBlocked(muzzle, root.Position, { c, myChar }) and not SA_anyVisible(c, muzzle) then
        wbSpot = SA_findWbSpot(root, muzzle)
    end
    if wbSpot and att then
        SA_wbPush = { att = att, orig = att.WorldCFrame }
        pcall(function() att.WorldCFrame = CFrame.new(wbSpot) end)
    end
    local quietCf
    if SA.quiet and not wbSpot then quietCf = SA_findQuietSpot(root, myChar) end
    local order
    if conf >= 0.75 or SA.part == "Head" then
        order = { "Head","UpperTorso","Torso","HumanoidRootPart","LowerTorso" }
    else order = { "UpperTorso","Torso","LowerTorso","HumanoidRootPart","Head" } end
    local fromPos = (SA_wbPush and SA_wbPush.att.WorldPosition) or (quietCf and quietCf.Position) or muzzle
    rp.FilterDescendantsInstances = { myChar, workspace.CurrentCamera }
    local candidates = {}
    for _, name in ipairs(order) do
        local part = c:FindFirstChild(name)
        if part and part:IsA("BasePart") then
            local aimPos = part.Position + off
            local dir = aimPos - fromPos
            local hit = workspace:Raycast(fromPos, dir, rp)
            if not hit or hit.Instance:IsDescendantOf(c) then
                table.insert(candidates, { pos = aimPos, dist = dir.Magnitude,
                    weight = BODY_WEIGHT[name] or 1, name = name })
            end
        end
    end
    local picked
    if #candidates > 0 then
        table.sort(candidates, function(a, b)
            if a.weight ~= b.weight then return a.weight > b.weight end
            return a.dist < b.dist
        end)
        picked = candidates[1].pos
    end
    if picked then
        if SA_lock.blendT > 0 and SA_lock.lastPos then
            local t = 1 - SA_lock.blendT
            local smooth = t * SA.blend
            picked = SA_lock.lastPos + (picked - SA_lock.lastPos) * smooth
            SA_lock.blendT = math.max(0, SA_lock.blendT - 0.16)
        end
        SA_lock.lastPos = picked
        return picked
    end
    local fallback
    for _, name in ipairs(order) do
        local part = c:FindFirstChild(name)
        if part and part:IsA("BasePart") then fallback = part.Position + off; break end
    end
    return fallback or (root.Position + off)
end

        pcall(function() SA_wbPush.att.WorldCFrame = SA_wbPush.orig end)
        SA_wbPush = nil
    end
end

local AntiAim = {}

function AntiAim.initialize(configuration)
    if type(configuration.fakepos) ~= "table" then configuration.fakepos = {} end
    local settings = configuration.fakepos
    local version = tonumber(settings.v) or 0
    if tonumber(settings.power) == nil or version < 3 then settings.power = 5 end
    if settings.vel == nil then settings.vel = true end
    if settings.md == nil then settings.md = true end
    if settings.side == nil or version < 4 then settings.side = true end
    if version < 5 then settings.mode, settings.dir, settings.smart, settings.power = "Desync", "Random", true, 10 end
    if settings.mode ~= "Desync" and settings.mode ~= "Prediction" then settings.mode = "Desync" end
    if type(settings.dir) ~= "string" then settings.dir = "Random" end
    if settings.smart == nil then settings.smart = true end
    settings.v, settings.on = 5, false
    return settings
end

function AntiAim.bindCharacter(state, character)
    local context, environment = state.context, state.context.environment
    state.character, state.active = character, false
    for _, connection in state.characterConnections do pcall(function() connection:Disconnect() end) end
    state.characterConnections = {}
    if character then
        local function onTool(tool)
            if tool:IsA("Tool") then state.toolAt = environment.os.clock() end
        end
        table.insert(state.characterConnections, context.connect(character.ChildAdded, onTool))
        table.insert(state.characterConnections, context.connect(character.ChildRemoved, onTool))
    end
    state.savedVelocity, state.spoofVelocity, state.savedDirection, state.spoofDirection = nil, nil, nil, nil
    state.savedFrame, state.spoofFrame = nil, nil
    state.root = character and character:FindFirstChild("HumanoidRootPart") or nil
    state.humanoid = character and character:FindFirstChildOfClass("Humanoid") or nil
end

function AntiAim.restore(state)
    state.active = false
    local root, humanoid = state.root, state.humanoid
    if state.savedFrame and state.spoofFrame and root and root.Parent then
        local current = root.CFrame
        if (current.Position - state.spoofFrame.Position).Magnitude < 0.01
            and current.LookVector:Dot(state.spoofFrame.LookVector) > 0.9999 then root.CFrame = state.savedFrame end
    end
    if state.spoofVelocity and state.savedVelocity and root and root.Parent then
        local current = root.AssemblyLinearVelocity
        local spoofDistance, originalDistance = (current - state.spoofVelocity).Magnitude, (current - state.savedVelocity).Magnitude
        if spoofDistance < 0.05 then root.AssemblyLinearVelocity = state.savedVelocity end
        if spoofDistance >= 0.05 and spoofDistance < originalDistance then
            local vector = state.context.environment.Vector3
            root.AssemblyLinearVelocity = vector.new(state.savedVelocity.X, math.min(current.Y, state.savedVelocity.Y), state.savedVelocity.Z)
            state.leakFix = (state.leakFix or 0) + 1
        end
    end
    if state.spoofDirection and state.savedDirection and humanoid and humanoid.Parent then
        if (humanoid.MoveDirection - state.spoofDirection).Magnitude < 0.05 then humanoid:Move(state.savedDirection, false) end
    end
    state.savedVelocity, state.spoofVelocity, state.savedDirection, state.spoofDirection = nil, nil, nil, nil
    state.savedFrame, state.spoofFrame = nil, nil
    state.restores += 1
end

function AntiAim.tryRestore(state)
    if not state.active then return end
    local ok = pcall(AntiAim.restore, state)
    if ok then return end
    state.active = false
    state.savedVelocity, state.spoofVelocity, state.savedDirection, state.spoofDirection = nil, nil, nil, nil
    state.savedFrame, state.spoofFrame = nil, nil
end

function AntiAim.ownedMover(state)
    local root = state.root
    if not root or not root.Parent then return false end
    for _, child in root:GetChildren() do
        if child.Name == "PulseFarmBV" or child.Name == "PulseFarmBG" or child:GetAttribute("PulseMine") then return true end
    end
    return false
end

function AntiAim.busy(state, now)
    local globals = state.context.environment.getgenv()
    if now < (tonumber(globals.__pulseShotHold) or 0) or now - (state.toolAt or 0) < 0.35 then return true end
    if globals.__pulseFlyActive or globals.__pulseManualFling or globals.__pulseTouchFlinging
        or globals.__pulseFlingTargetChar or globals.__pulseSpinActive then return true end
    if state.context.configuration.autoFlingSheriff then return true end
    if globals.__pulseInvis and globals.__pulseInvis.active then return true end
    if (tonumber(globals.__pulseBombJumpUntil) or 0) > now then return true end
    local farm = globals.__pulseFarmAPI
    if farm and farm.getEnabled and farm.getEnabled() then return true end
    return state.hasMover
end

function AntiAim.threat(state, position)
    local context = state.context
    local roles = context.roles()
    if type(roles) ~= "table" then return nil end
    local own = roles[context.player.Name]
    local murderer = own ~= nil and tostring(own.Role) == "Murderer"
    local closest, distance
    for _, player in context.players:GetPlayers() do
        if player == context.player then continue end
        local role = roles[player.Name]
        if not role or role.Dead == true or role.Killed == true then continue end
        local name = tostring(role.Role)
        if murderer and name ~= "Sheriff" and name ~= "Hero" then continue end
        if not murderer and name ~= "Murderer" then continue end
        local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
        if not root then continue end
        local current = (root.Position - position).Magnitude
        if not distance or current < distance then closest, distance = root, current end
    end
    return closest
end

function AntiAim.spoofVector(state, now, power)
    local vector, settings = state.context.environment.Vector3, state.cfg
    local function random() return math.random() * 2 - 1 end
    if settings.dir == "Underground" then return vector.new(random() * power * 0.15, -power, random() * power * 0.15) end
    if settings.dir == "Sky" then return vector.new(random() * power * 0.15, power, random() * power * 0.15) end
    if settings.dir == "Spin" then
        state.angle = (state.angle + 0.9 + math.random() * 0.6) % (2 * math.pi)
        return vector.new(math.cos(state.angle) * power, random() * power * 0.35, math.sin(state.angle) * power)
    end
    if settings.dir == "Jitter" then
        if not state.jitterDirection or now >= state.jitterUntil then
            local direction = vector.new(random(), random() * 0.4, random())
            state.jitterDirection = direction.Magnitude > 0.1 and direction.Unit or vector.xAxis
            state.jitterUntil = now + 0.25 + math.random() * 0.25
        end
        state.flip = -state.flip
        return state.jitterDirection * (power * 1.6 * state.flip)
    end
    return vector.new((math.random() < 0.5 and -1 or 1) * power, random() * power,
        (math.random() < 0.5 and -1 or 1) * power)
end

function AntiAim.nearCoin(state, position, now)
    local context = state.context
    if now - state.coinAt > 1 or not (state.coinContainer and state.coinContainer.Parent) then
        state.coinAt, state.coinContainer = now, nil
        for _, child in context.workspace:GetChildren() do
            local container = child:FindFirstChild("CoinContainer")
            if container then state.coinContainer = container; break end
        end
        if state.coinContainer then state.coinParams.FilterDescendantsInstances = {state.coinContainer} end
    end
    if not state.coinContainer then return false end
    return #context.workspace:GetPartBoundsInRadius(position, state.coinR or 14, state.coinParams) > 0
end

function AntiAim.shouldPause(state, character, position, now)
    if now - state.pauseAt < 0.08 then return state.pauseValue end
    state.pauseAt = now
    local context, environment = state.context, state.context.environment
    local knife, gun = character:FindFirstChild("Knife"), character:FindFirstChild("Gun")
    local pause = (knife or gun) and context.inputService:IsMouseButtonPressed(environment.Enum.UserInputType.MouseButton1) or false
    if knife and not pause then
        local globals = environment.getgenv()
        pause = globals.__pulseKillAuraOn or (globals.__pulseGetKillAll and globals.__pulseGetKillAll()) or false
        if not pause then
            for _, player in context.players:GetPlayers() do
                if player == context.player then continue end
                local target = player.Character
                local root = target and target:FindFirstChild("HumanoidRootPart")
                local humanoid = target and target:FindFirstChildOfClass("Humanoid")
                if root and humanoid and humanoid.Health > 0 and (root.Position - position).Magnitude < (state.stabR or 16) then
                    pause = true; break
                end
            end
        end
    end
    if not pause then
        local ok, near = pcall(AntiAim.nearCoin, state, position, now)
        pause = ok and near or false
    end
    state.pauseValue = pause
    return pause
end

function AntiAim.desync(state, character, position, now)
    local context, settings = state.context, state.cfg
    local roles = context.roles()
    local role = roles and roles[context.player.Name]
    if not role or role.Dead == true or role.Killed == true then state.mode = "idle"; return end
    if settings.smart ~= false and AntiAim.shouldPause(state, character, position, now) then state.mode = "pause"; return end
    local vector = context.environment.Vector3
    local power = 32767 * math.clamp(tonumber(settings.power) or 10, 1, 10) / 10
    local spoof = AntiAim.spoofVector(state, now, power)
    state.active = true
    state.savedVelocity = state.root.AssemblyLinearVelocity
    state.spoofVelocity = vector.new(math.clamp(spoof.X, -32767, 32767), math.clamp(spoof.Y, -32767, 32767),
        math.clamp(spoof.Z, -32767, 32767))
    state.root.AssemblyLinearVelocity = state.spoofVelocity
    if context.environment.sethiddenproperty then pcall(context.environment.sethiddenproperty, state.root, "NetworkIsSleeping", false) end
    if settings.md ~= false then
        if not state.direction or now >= state.directionUntil then
            local angle = math.random() * 2 * math.pi
            state.direction = vector.new(math.cos(angle), 0, math.sin(angle))
            state.directionUntil = now + 0.15 + math.random() * 0.3
        end
        state.savedDirection, state.spoofDirection = state.humanoid.MoveDirection, state.direction
        state.humanoid:Move(state.spoofDirection, false)
    end
    state.mode, state.desyncAt, state.frames = "desync", now, state.frames + 1
end

function AntiAim.predict(state, position, now)
    local context, settings = state.context, state.cfg
    local environment, vector = context.environment, context.environment.Vector3
    state.mode = "prediction"
    if now >= state.threatAt then
        state.threatAt = now + 0.25
        local ok, threat = pcall(AntiAim.threat, state, position)
        state.threat = ok and threat or nil
    end
    if not state.threat or not state.threat.Parent then return end
    local delta = position - state.threat.Position
    delta = vector.new(delta.X, 0, delta.Z)
    if delta.Magnitude < 0.5 then return end
    local direction = delta.Unit
    local side = vector.new(-direction.Z, 0, direction.X)
    local velocity, frame = state.root.AssemblyLinearVelocity, state.root.CFrame
    local horizontal = vector.new(velocity.X, 0, velocity.Z)
    local lateral = horizontal:Dot(side)
    local facing = frame.LookVector:Dot(side) >= 0 and side or -side
    if math.abs(lateral) > 3 then
        state.side = lateral > 0 and -1 or 1
        state.sideUntil = now + 0.25 + math.random() * 0.35
    end
    if math.abs(lateral) <= 3 and settings.side ~= false and now >= state.sideUntil then
        state.side = facing:Dot(side) > 0 and 1 or -1
    end
    if math.abs(lateral) <= 3 and settings.side == false and now >= state.sideUntil then
        state.side = math.random() < 0.5 and -1 or 1
        state.sideUntil = now + 0.6 + math.random() * 1.2
    end
    local spoofDirection = side * state.side
    local speed = math.max(10 + math.clamp(tonumber(settings.power) or 5, 1, 10) * 9, horizontal.Magnitude)
    state.active = true
    if settings.vel ~= false then
        state.savedVelocity = velocity
        state.spoofVelocity = vector.new(spoofDirection.X * speed, velocity.Y, spoofDirection.Z * speed)
        state.root.AssemblyLinearVelocity = state.spoofVelocity
    end
    if settings.md ~= false then
        state.savedDirection, state.spoofDirection = state.humanoid.MoveDirection, spoofDirection
        state.humanoid:Move(spoofDirection, false)
    end
    if settings.side ~= false then
        local fake = environment.CFrame.lookAt(frame.Position, frame.Position + facing)
        if fake then
            state.savedFrame, state.spoofFrame = frame, fake
            state.root.CFrame = fake
            state.sideFrames, state.sideAt, state.fakeCF = (state.sideFrames or 0) + 1, now, fake
        end
    end
    state.frames += 1
end

function AntiAim.step(state)
    local context, settings = state.context, state.cfg
    if state.active or not settings.on then return end
    if settings.mode == "Prediction" and settings.vel == false and settings.md == false and settings.side == false then return end
    local character = context.player.Character
    if character ~= state.character then AntiAim.bindCharacter(state, character) end
    if not (state.root and state.root.Parent) then state.root = character and character:FindFirstChild("HumanoidRootPart") or nil end
    if not (state.humanoid and state.humanoid.Parent) then state.humanoid = character and character:FindFirstChildOfClass("Humanoid") or nil end
    local root, humanoid = state.root, state.humanoid
    if not root or not humanoid or humanoid.Health <= 0 or humanoid.Sit or state.invalidStates[humanoid:GetState()] then return end
    local now = context.environment.os.clock()
    if now >= state.moverAt then state.moverAt, state.hasMover = now + 0.1, AntiAim.ownedMover(state) end
    if AntiAim.busy(state, now) then state.mode = "busy"; return end
    if settings.mode ~= "Prediction" then AntiAim.desync(state, character, root.Position, now); return end
    AntiAim.predict(state, root.Position, now)
end

function AntiAim.install(context)
    local environment, globals = context.environment, context.environment.getgenv()
    local settings = AntiAim.initialize(context.configuration)
    local invalid = environment.Enum.HumanoidStateType
    local state = {context = context, cfg = settings, frames = 0, restores = 0, side = 1, sideUntil = 0, threatAt = 0,
        angle = 0, flip = 1, jitterUntil = 0, directionUntil = 0, pauseAt = 0, pauseValue = false, coinAt = -10,
        moverAt = 0, hasMover = false, active = false, characterConnections = {}, invalidStates = {}}
    for _, name in {"Dead", "Seated", "Climbing", "Physics", "PlatformStanding", "Ragdoll", "FallingDown", "Swimming", "GettingUp"} do
        state.invalidStates[invalid[name]] = true
    end
    state.coinParams = environment.OverlapParams.new()
    state.coinParams.FilterType = environment.Enum.RaycastFilterType.Include
    globals.__pulseFakePos = {active = false, off = environment.Vector3.zero}
    globals.__pulseAntiAim = state

    context.connect(context.runService.Heartbeat, function()
        AntiAim.tryRestore(state)
        if settings.on then environment.task.defer(function()
            if not pcall(AntiAim.step, state) then AntiAim.tryRestore(state) end
        end) end
    end)
    context.connect(context.runService.PreSimulation, function() AntiAim.tryRestore(state) end)
    pcall(function() context.runService:UnbindFromRenderStep("PulseFakePos") end)
    pcall(function() context.runService:BindToRenderStep("PulseFakePos", environment.Enum.RenderPriority.First.Value, function() AntiAim.tryRestore(state) end) end)
    context.connect(context.player.CharacterAdded, function(character) AntiAim.bindCharacter(state, character) end)
    if context.player.Character then AntiAim.bindCharacter(state, context.player.Character) end

    return state, function()
        AntiAim.tryRestore(state)
        settings.on = false
        for _, connection in state.characterConnections do pcall(function() connection:Disconnect() end) end
        pcall(function() context.runService:UnbindFromRenderStep("PulseFakePos") end)
        globals.__pulseFakePos.active, globals.__pulseFakePos.off = false, environment.Vector3.zero
        globals.__pulseAntiAim = nil
    end
end

-- [AIM HOOK] — __namecall on FireServer with fallback
-- ============================================================
local aim_hook_installed = false
local aim_namecall_ok = false
local og_namecall = nil

local function is_gun_remote(re)
    local c = LP.Character
    if not c then return false end
    local gun = c:FindFirstChild("Gun") or c:FindFirstChild("Weapon") or c:FindFirstChild("Revolver")
    if gun then
        if re:IsDescendantOf(gun) or re:IsDescendantOf(c) then return true end
    end
    local n = re.Name:lower()
    if n:find("fire") or n:find("shoot") or n:find("gun") then return true end
    local p = re.Parent
    if p then
        local pn = p.Name:lower()
        if pn:find("gun") or pn:find("fire") or pn:find("shoot") then return true end
    end
    return false
end

install_aim_hook = function()
    if aim_hook_installed then return end
    aim_hook_installed = true

    if type(hookmetamethod) ~= "function" or type(getnamecallmethod) ~= "function" then
        getgenv().__aim_no_namecall = true
        return
    end

    local ok = pcall(function()
        og_namecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
            if getnamecallmethod() == "FireServer" and SA.enabled then
                local okg, is_gun = pcall(is_gun_remote, self)
                if okg and is_gun then
                    local okpt, pt = pcall(resolve)
                    if okpt and pt then
                        local args = table.pack(...)
                        local has_cf = false
                        for i = 1, args.n do
                            if typeof(args[i]) == "CFrame" then
                                args[i] = CFrame.new(pt)
                                has_cf = true
                            end
                        end
                        if has_cf then
                            getgenv().__pulseShotHold = os.clock() + 0.35
                            local r = og_namecall(self, table.unpack(args, 1, args.n))
                            if SA_wbPush then
                                task.defer(restore_wb)
                                task.delay(0.08, restore_wb)
                            end
                            return r
                        end
                    end
                end
            end
            return og_namecall(self, ...)
        end))
        aim_namecall_ok = true
    end)

    if not ok then
        getgenv().__aim_no_namecall = true
    end
end

install_gun_remote_hooks = function(tool)
    if not tool or not tool:IsA("Tool") then return end
    if aim_namecall_ok then return end -- namecall covers it
    for _, r in ipairs(tool:GetDescendants()) do
        if r:IsA("RemoteEvent") and not r:GetAttribute("__aimh") then
            r:SetAttribute("__aimh", true)
            local old = r.FireServer
            if type(old) ~= "function" then continue end
            pcall(function()
                r.FireServer = function(self, ...)
                    if SA.enabled then
                        local okpt, pt = pcall(resolve)
                        if okpt and pt then
                            local args = table.pack(...)
                            local has_cf = false
                            for i = 1, args.n do
                                if typeof(args[i]) == "CFrame" then
                                    args[i] = CFrame.new(pt)
                                    has_cf = true
                                end
                            end
                            if has_cf then
                                getgenv().__pulseShotHold = os.clock() + 0.35
                                local r2 = old(self, table.unpack(args, 1, args.n))
                                if SA_wbPush then
                                    task.defer(restore_wb)
                                    task.delay(0.08, restore_wb)
                                end
                                return r2
                            end
                        end
                    end
                    return old(self, ...)
                end
            end)
        end
    end
end

ka_tick = function()
    if not KA.on then return end
    local now = os.clock()
    if now - KA.last < KA.cooldown then return end
    local c = LP.Character; if not c then return end
    local knife = c:FindFirstChild("Knife")
    if not knife then
        local bp = LP:FindFirstChildOfClass("Backpack")
        knife = bp and bp:FindFirstChild("Knife")
        if knife then
            local h = c:FindFirstChildOfClass("Humanoid")
            if h then pcall(function() h:EquipTool(knife) end) end
        end
        return
    end
    local myp = c:FindFirstChild("HumanoidRootPart"); if not myp then return end
    local events = knife:FindFirstChild("Events")
    local stabbed = events and events:FindFirstChild("KnifeStabbed")
    local touched = events and events:FindFirstChild("HandleTouched")
    if not stabbed or not touched then return end
    local fired = false
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and alive(plr) then
            local tchar = plr.Character
            local tpart = tchar and (tchar:FindFirstChild("HumanoidRootPart") or tchar:FindFirstChild("Head"))
            if tpart and (tpart.Position - myp.Position).Magnitude <= KA.dist then
                if not fired then pcall(function() stabbed:FireServer() end); fired = true end
                pcall(function() touched:FireServer(tpart) end)
            end
        end
    end
    if fired then KA.last = now end
end

local function to_pos(v)
    if typeof(v) == "Vector3" then return v end
    if typeof(v) == "CFrame" then return v.Position end
    if typeof(v) == "Instance" then
        if v:IsA("Attachment") then return v.WorldPosition end
        if v:IsA("BasePart") then return v.Position end
    end
end

    if TRACE.conn then return end

        local m = Replicated:FindFirstChild("ClientServices")

            m = m:FindFirstChild("WeaponService")
            if m then ev = m:FindFirstChild("GunFired") end
        end

        end)
    end)

bt_destroy = function()
    safe_destroy(BT.model); BT.model = nil
    BT.cache = {}; BT.hist = {}; BT.first = 1; BT.n = 0
end
bt_build = function()
    bt_destroy()
    local char = LP.Character; if not char then return end
    local rhrp = char:FindFirstChild("HumanoidRootPart"); if not rhrp then return end
    local prevArch = char.Archivable
    char.Archivable = true
    local ok, m = pcall(function() return char:Clone() end)
    char.Archivable = prevArch
    if not ok or not m then return end
    local rparts = {}
    for _, o in char:GetDescendants() do
        if o:IsA("BasePart") then rparts[#rparts+1] = o end
    end
    local ci = 0
    for _, o in m:GetDescendants() do
        if o:IsA("Script") or o:IsA("LocalScript") or o:IsA("ModuleScript")
        or o:IsA("Decal") or o:IsA("Texture") or o:IsA("ParticleEmitter") or o:IsA("Beam")
        or o:IsA("Trail") or o:IsA("Light") or o:IsA("Sound") then
            pcall(function() o:Destroy() end)
        elseif o:IsA("BasePart") then
            o.Anchored = true; o.CanCollide = false; o.CanQuery = false
            o.CanTouch = false; o.Massless = true; o.CastShadow = false
            if o.Name == "HumanoidRootPart" then o.Transparency = 1
            else
                o.Material = Enum.Material.ForceField
                o.Color = FC.bt
                o.Transparency = SZ.bt
            end
            ci = ci + 1
            if rparts[ci] then BT.cache[#BT.cache+1] = {o, rparts[ci]} end
        end
    end
    local h2 = m:FindFirstChildOfClass("Humanoid")
    if h2 then pcall(function() h2:Destroy() end) end
    m.Name = "\0"; m.Parent = workspace
    BT.model = m
end
bt_update = function()
    if not BT.on then return end
    local char = LP.Character
    local rhrp = char and char:FindFirstChild("HumanoidRootPart"); if not rhrp then return end
    if not BT.model or not BT.model.Parent then bt_build(); if not BT.model then return end end
    local now = os.clock()
    local base_cf = rhrp.CFrame
    local slotIdx
    if BT.n < BT.cap then BT.n = BT.n + 1; slotIdx = BT.n
    else BT.first = (BT.first % BT.cap) + 1; slotIdx = BT.first end
    local slot = BT.hist[slotIdx]
    if not slot then slot = {}; BT.hist[slotIdx] = slot end
    slot[1] = now; slot[2] = base_cf
    if now - BT.lastping >= 0.2 then
        BT.lastping = now
        local ok, val = pcall(function() return Stats.Network.ServerStatsItem["Data Ping"]:GetValue()/1000 end)
        BT.ping = math.clamp((ok and val) or 0.15, 0.05, 0.6)
    end
    local target = now - BT.ping
    local cf = base_cf
    for i = 0, BT.n - 1 do
        local idx = ((BT.first + i - 1) % BT.cap) + 1
        local s = BT.hist[idx]
        if s and s[1] and s[1] <= target then cf = s[2]; break end
    end
    while BT.n > 0 do
        local s = BT.hist[BT.first]
        if not s or s[1] >= now - 1 then break end
        BT.first = (BT.first % BT.cap) + 1
        BT.n = BT.n - 1
    end
    local baseInv = rhrp.CFrame:Inverse()
    for i = 1, #BT.cache do
        local cp, rpv = BT.cache[i][1], BT.cache[i][2]
        if cp and cp.Parent and rpv and rpv.Parent then
            cp.CFrame = cf * (baseInv * rpv.CFrame)
        end
    end
end
bt_conn = RS.Heartbeat:Connect(function() if BT.on then pcall(bt_update) end end)

local function flingSetup(hrp, hum2)
    local old = {}
    old.cf = hrp.CFrame
    old.seated = hum2:GetStateEnabled(Enum.HumanoidStateType.Seated)
    old.fall = workspace.FallenPartsDestroyHeight
    old.camSubj = workspace.CurrentCamera and workspace.CurrentCamera.CameraSubject
    pcall(function() hum2:SetStateEnabled(Enum.HumanoidStateType.Seated, false) end)
    pcall(function() workspace.FallenPartsDestroyHeight = 0/0 end)
    return old
end
local function flingTeardown(hrp, hum2, old)
    pcall(function() hum2:SetStateEnabled(Enum.HumanoidStateType.Seated, old.seated) end)
    pcall(function() workspace.FallenPartsDestroyHeight = old.fall end)
    if workspace.CurrentCamera and old.camSubj then
        pcall(function() workspace.CurrentCamera.CameraSubject = old.camSubj end)
    end
    if hrp.Parent then
        pcall(function()
            hrp.CFrame = old.cf
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
            hum2:ChangeState(Enum.HumanoidStateType.GettingUp)
        end)
    end
end

local function flingOrbit(hrp, thum, thrp, dur)
    local ang = 0
    local t0 = os.clock()
    while os.clock() - t0 < dur and hrp.Parent and thrp.Parent do
        local tv = thrp.Velocity
        if tv.Magnitude > 50 and thum then tv = thum.MoveDirection * thum.WalkSpeed end
        local lead = thrp.Position + tv * 0.08
        ang = ang + 165
        local r = ang % 360
        local off
        if r < 90 then off = Vector3.new(0, 2.5, 0)
        elseif r < 180 then off = Vector3.new(0, -2.5, 0)
        elseif r < 270 then off = Vector3.new(2.5, 0, 0)
        else off = Vector3.new(-2.5, 0, 0) end
        pcall(function()
            hrp.CFrame = CFrame.new(lead + off) * CFrame.Angles(math.rad(ang), 0, 0)
            hrp.AssemblyLinearVelocity = Vector3.new(FLING.force, FLING.force * 10, FLING.force)
            hrp.AssemblyAngularVelocity = Vector3.new(9e9, 9e9, 9e9)
        end)
        RS.Heartbeat:Wait()
    end
end

local function flingSlam(hrp, thum, thrp, dur)
    local t0 = os.clock()
    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9); bv.P = 1e5; bv.Parent = hrp
    local bav = Instance.new("BodyAngularVelocity")
    bav.MaxTorque = Vector3.new(9e9, 9e9, 9e9); bav.P = 1e5; bav.Parent = hrp
    while os.clock() - t0 < dur and hrp.Parent and thrp.Parent do
        local dir = (thrp.Position - hrp.Position)
        if dir.Magnitude < 1 then dir = Vector3.new(0, 1, 0) end
        dir = dir.Unit
        pcall(function()
            bv.Velocity = dir * FLING.force + Vector3.new(0, FLING.force * 0.5, 0)
            bav.AngularVelocity = Vector3.new(FLING.force, FLING.force, FLING.force)
        end)
        RS.Heartbeat:Wait()
    end
    safe_destroy(bv); safe_destroy(bav)
end

local function flingSpin(hrp, thum, thrp, dur)
    local t0 = os.clock()
    local bav = Instance.new("BodyAngularVelocity")
    bav.MaxTorque = Vector3.new(9e9, 9e9, 9e9); bav.Parent = hrp
    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9); bv.Parent = hrp
    local spin = 0
    while os.clock() - t0 < dur and hrp.Parent and thrp.Parent do
        spin = spin + 800
        local dir = (thrp.Position - hrp.Position)
        if dir.Magnitude < 1 then dir = Vector3.new(0, 1, 0) end
        dir = dir.Unit
        pcall(function()
            bav.AngularVelocity = Vector3.new(math.rad(spin) % (2*math.pi), spin, 0)
            bv.Velocity = dir * (FLING.force * 0.5)
            hrp.RotVelocity = Vector3.new(9e9, 9e9, 9e9)
        end)
        RS.Heartbeat:Wait()
    end
    safe_destroy(bav); safe_destroy(bv)
end

local function flingAnchor(hrp, thum, thrp, dur)
    local t0 = os.clock()
    local wasAnchored = hrp.Anchored
    local wasMassless = hrp.Massless
    hrp.Anchored = false; hrp.Massless = true
    local bav = Instance.new("BodyAngularVelocity")
    bav.MaxTorque = Vector3.new(9e9, 9e9, 9e9); bav.Parent = hrp
    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9); bv.Parent = hrp
    local angle = 0
    while os.clock() - t0 < dur and hrp.Parent and thrp.Parent do
        angle = angle + 240
        local r = math.rad(angle)
        local r2 = 8 + 4 * math.sin(r * 1.5)
        local offset = Vector3.new(math.cos(r) * r2, 3 + 2 * math.sin(r * 2), math.sin(r) * r2)
        pcall(function()
            hrp.CFrame = CFrame.new(thrp.Position + offset) * CFrame.Angles(r * 3, r, 0)
            hrp.AssemblyLinearVelocity = Vector3.new(FLING.force, FLING.force * 2, FLING.force)
            hrp.AssemblyAngularVelocity = Vector3.new(9e9, 9e9, 9e9)
            bav.AngularVelocity = Vector3.new(9e9, 9e9, 9e9)
            bv.Velocity = Vector3.new(FLING.force, FLING.force * 3, FLING.force)
        end)
        RS.Heartbeat:Wait()
    end
    hrp.Anchored = wasAnchored; hrp.Massless = wasMassless
    safe_destroy(bav); safe_destroy(bv)
end

local function flingCore(thrp, thum, dur)
    local c = LP.Character
    local hrp = c and c:FindFirstChild("HumanoidRootPart"); if not hrp then return false end
    local hum2 = c:FindFirstChildOfClass("Humanoid"); if not hum2 then return false end
    if thum and thum.Sit then return false end
    getgenv().__pulseFlingTargetChar = thrp
    local old = flingSetup(hrp, hum2)
    local d = dur or FLING.duration
    if FLING.mode == "orbit" then flingOrbit(hrp, thum, thrp, d)
    elseif FLING.mode == "slam" then flingSlam(hrp, thum, thrp, d)
    elseif FLING.mode == "spin" then flingSpin(hrp, thum, thrp, d)
    elseif FLING.mode == "anchor" then flingAnchor(hrp, thum, thrp, d)
    else flingOrbit(hrp, thum, thrp, d) end
    flingTeardown(hrp, hum2, old)
    getgenv().__pulseFlingTargetChar = nil
    return true
end

        TROLL.flinging = false; FLING.active = false
    end)
end

auto_shoot_tick = function()
    if not AUTO.on then return end
    local c = LP.Character; if not c then return end
    local h = c:FindFirstChildOfClass("Humanoid")
    if not h or h.Health <= 0 then return end
    local gun = c:FindFirstChild("Gun"); if not gun then return end
    local now = os.clock()
    if now - AUTO.last < AUTO.delay then return end
    local cam = workspace.CurrentCamera; if not cam then return end
    local myPos, look = cam.CFrame.Position, cam.CFrame.LookVector
    local best, bestAng = nil, AUTO.fov
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and alive(p) then
            if not AUTO.murderer_only or cachedRole(p) == "m" then
                local pc = p.Character
                local head = pc and pc:FindFirstChild("Head")
                if head then
                    local toHead = head.Position - myPos
                    if toHead.Magnitude > 1 then
                        local ang = math.deg(math.acos(math.clamp(look:Dot(toHead.Unit), -1, 1)))
                        if ang < bestAng then
                            if not SA_rayBlocked(myPos, head.Position, { pc, c }) then
                                best = p; bestAng = ang
                            end
                        end
                    end
                end
            end
        end
    end
    if not best then return end
    if h:FindFirstChildOfClass("Tool") ~= gun then
        pcall(function() h:EquipTool(gun) end)
        AUTO.last = now; return
    end
    pcall(function()
        if mouse1click then mouse1click()
        elseif mouse1press and mouse1release then mouse1press(); task.wait(0.02); mouse1release() end
    end)
    AUTO.last = now
end

end

            p.Parent = boFolder
            local vel = Vector3.new((math.random()-0.5)*6, 3+math.random()*3, (math.random()-0.5)*6)
            local life = 0
            local rp2 = RaycastParams.new()
            rp2.FilterDescendantsInstances = { c }
            rp2.FilterType = Enum.RaycastFilterType.Exclude
            local cn
            cn = RS.Heartbeat:Connect(function(dt)
                life = life + dt
                if not p.Parent then safe_disconnect(cn); return end
                vel = vel + Vector3.new(0, -30*dt, 0)
                local np = p.Position + vel*dt
                if vel.Y < 0 then
                    local ray = workspace:Raycast(p.Position, Vector3.new(0,-6,0), rp2)
                    if ray and (p.Position.Y - ray.Position.Y) < 0.7 then
                        np = Vector3.new(np.X, ray.Position.Y + baseSize/2 + 0.1, np.Z)
                        vel = Vector3.new(vel.X, -vel.Y*0.6, vel.Z)
                    end
                end
                p.Position = np
                if life >= 3 then safe_disconnect(cn); safe_destroy(p) end
            end)
        end
    end)

local ESP_TXT = { Murderer=Color3.fromRGB(255,60,60), Sheriff=Color3.fromRGB(70,150,255), Hero=Color3.fromRGB(70,150,255) }
local espTxtTags = {}
local function roleLabel(pl)
    refreshRoleCache()
    local r = roleCache[pl.Name]
    if r then return r end
    local g = getRole(pl)
    if g == "m" then return "Murderer" end
    if g == "s" then return "Sheriff" end
    return nil
end
local function attachRoleTag(pl, head, role)
    local bb = espTxtTags[pl]
    if not bb or not bb.Parent then
        bb = Instance.new("BillboardGui")
        bb.Name="RoleTag"; bb.Size=UDim2.new(0,120,0,28)
        bb.StudsOffset=Vector3.new(0,2.6,0); bb.AlwaysOnTop=true; bb.LightInfluence=0
        local l = Instance.new("TextLabel", bb)
        l.Name="L"; l.Size=UDim2.new(1,0,1,0); l.BackgroundTransparency=1
        l.Font=Enum.Font.GothamBold; l.TextScaled=true
        l.TextStrokeTransparency=0; l.TextStrokeColor3=Color3.new(0,0,0)
        bb.Parent = head; espTxtTags[pl] = bb
    end
    if bb.Parent ~= head then bb.Parent = head end
    local l = bb:FindFirstChild("L")
    if l then l.Text = role; l.TextColor3 = ESP_TXT[role] or Color3.new(1,1,1) end
end
local function detachRoleTag(pl)
    local bb = espTxtTags[pl]
    safe_destroy(bb)
    espTxtTags[pl] = nil
end
local function updateRoleTags()
    for _, pl in ipairs(Players:GetPlayers()) do
        if pl ~= LP then
            local c = pl.Character
            local h = c and c:FindFirstChild("Head")
            local r = c and roleLabel(pl) or nil
            if h and r then attachRoleTag(pl, h, r) else detachRoleTag(pl) end
        else detachRoleTag(pl) end
    end
end

-- Jump FX
local function mkJump()
    safe_disconnect(jumpC); jumpC = nil
    local c = LP.Character; if not c then return end
    local h = c:FindFirstChildOfClass("Humanoid"); if not h then return end
    jumpC = h.Jumping:Connect(function(active)
        if not active then return end
        local hrp = HRP(); if not hrp then return end
        local pos = hrp.Position - Vector3.new(0,2.5,0)
        local col = FC.jump
        local function ring()
            local p = Instance.new("Part")
            p.Size=Vector3.new(1,1,1); p.Anchored=true
            p.CanCollide=false; p.CanQuery=false; p.CanTouch=false; p.CastShadow=false
            p.Material=Enum.Material.Neon; p.Color=col; p.Transparency=0.1
            p.CFrame = CFrame.new(pos + Vector3.new(0,0.7,0)) * CFrame.Angles(math.rad(90),0,0)
            local m = Instance.new("SpecialMesh", p)
            m.MeshId="rbxassetid://3270017"
            m.Scale=Vector3.new(SZ.jump,SZ.jump,0.3)
            p.Parent = workspace
            local ts = 0
            local cn
            cn = RS.Heartbeat:Connect(function(dt)
                ts = ts + dt; local k = ts/0.5
                if k >= 1 or not p.Parent then safe_disconnect(cn); safe_destroy(p); return end
                local sz = SZ.jump + k*6
                m.Scale = Vector3.new(sz,sz,0.3*(1-k*0.5))
                p.Transparency = 0.1 + k*0.9
            end)
        end
        if jStyle == "Classic" then ring()
        elseif jStyle == "Double" then ring(); task.delay(0.15, ring)
        elseif jStyle == "Ripple" then for i = 1, 3 do task.delay(i*0.08, ring) end
        elseif jStyle == "Particles" then
            ring()
            local a = Instance.new("Attachment", workspace.Terrain)
            a.WorldPosition = pos
            local pe = Instance.new("ParticleEmitter", a)
            pe.Texture="rbxasset://textures/particles/sparkles_main.dds"
            pe.Rate=0; pe.Lifetime=NumberRange.new(0.6,1)
            pe.Speed=NumberRange.new(6,10); pe.SpreadAngle=Vector2.new(180,180)
            pe.Size=NumberSequence.new({NumberSequenceKeypoint.new(0,0.8),NumberSequenceKeypoint.new(1,0)})
            pe.Transparency=NumberSequence.new({NumberSequenceKeypoint.new(0,0.1),NumberSequenceKeypoint.new(1,1)})
            pe.Color=ColorSequence.new(col,LT(col))
            pe.LightEmission=1; pe.LightInfluence=0; pe.Drag=3
            pe:Emit(35)
            task.delay(1.2, function() safe_destroy(a) end)
        else ring() end
    end)
end

-- Skin
local function applySkinAll()
    local c = LP.Character; if not c then return end
    local h = c:FindFirstChild("Head")
    if h then
        if fl.headless then
            pcall(function()
                h.Transparency = 1
                local d = h:FindFirstChildOfClass("Decal"); if d then d.Transparency = 1 end
            end)
        else
            pcall(function()
                h.Transparency = 0
                local d = h:FindFirstChildOfClass("Decal"); if d then d.Transparency = 0 end
            end)
        end
    end
end

-- Kill All
local function killAll()
    local c = LP.Character; if not c then MakeNotif("StarWare", "No character", 2); return end
    local knife = c:FindFirstChild("Knife")
    if not knife then
        local bp = LP:FindFirstChildOfClass("Backpack")
        knife = bp and bp:FindFirstChild("Knife")
        if knife then
            local h = c:FindFirstChildOfClass("Humanoid")
            if h then pcall(function() h:EquipTool(knife) end) end
            task.wait(0.3); knife = c:FindFirstChild("Knife")
        end
    end
    if not knife then MakeNotif("StarWare", "Need knife", 2); return end
    local events = knife:FindFirstChild("Events")
    local stabbed = events and events:FindFirstChild("KnifeStabbed")
    local touched = events and events:FindFirstChild("HandleTouched")
    if not stabbed or not touched then MakeNotif("StarWare", "No knife events", 2); return end
    local myp = c:FindFirstChild("HumanoidRootPart"); if not myp then return end
    local orig = myp.CFrame
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and alive(plr) then
            local tchar = plr.Character
            local tpart = tchar and (tchar:FindFirstChild("HumanoidRootPart") or tchar:FindFirstChild("Head"))
            if tpart then
                pcall(function() myp.CFrame = tpart.CFrame + Vector3.new(0,0,2) end)
                task.wait(0.1)
                pcall(function() stabbed:FireServer() end)
                pcall(function() touched:FireServer(tpart) end)
                task.wait(0.05)
            end
        end
    end
    pcall(function() myp.CFrame = orig end)
end
