local function makeSetting(initial, callback)
	local value = initial

-- ESP System -- PlayerData
local playerData = {}
local playerDataReady = false

-- Проверка, жив ли игрок на самом деле
local function isAlive(plr)
	if not plr or not plr.Parent then return false end
	
	if playerDataReady and playerData[plr.Name] then
		local data = playerData[plr.Name]
		if data.Dead or data.Killed then
			return false
		end
	end
	
	local char = plr.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	return hum ~= nil and hum.Health > 0
end

-- Проверка наличия оружия в руках или рюкзаке
local function hasTool(plr, toolName)
	local bp = plr:FindFirstChild("Backpack")
	if bp and bp:FindFirstChild(toolName) then return true end
	local char = plr.Character
	if char and char:FindFirstChild(toolName) then return true end
	return false
end

-- Поиск Мардера
local function findMurderer()
	if playerDataReady then
		for playerName, data in pairs(playerData) do
			if not data.Dead and not data.Killed and data.Role == "Murderer" then
				local plr = Players:FindFirstChild(playerName)
				if plr and isAlive(plr) then return plr end
			end
		end
	end

	for _, plr in ipairs(Players:GetPlayers()) do
		if isAlive(plr) and hasTool(plr, "Knife") then
			return plr
		end
	end
	return nil
end

-- Поиск Шерифа или Героя
local function findSheriff()
	if playerDataReady then
		for playerName, data in pairs(playerData) do
			if not data.Dead and not data.Killed and (data.Role == "Sheriff" or data.Role == "Hero") then
				local plr = Players:FindFirstChild(playerName)
				if plr and isAlive(plr) then return plr end
			end
		end
	end

	for _, plr in ipairs(Players:GetPlayers()) do
		if isAlive(plr) and hasTool(plr, "Gun") then
			return plr
		end
	end
	return nil
end

-- Умное обновление кэша (вызывайте в цикле отрисовки или при проверке)
local function updateRoleCache()
	if not isAlive(cachedMurderer) then
		cachedMurderer = findMurderer()
	end
	
	-- Для шерифа дополнительно проверяем, что он всё ещё жив и не потерял роль
	if not isAlive(cachedSheriff) then
		cachedSheriff = findSheriff()
	elseif playerDataReady and playerData[cachedSheriff.Name] then
		local role = playerData[cachedSheriff.Name].Role
		if role ~= "Sheriff" and role ~= "Hero" and not hasTool(cachedSheriff, "Gun") then
			cachedSheriff = findSheriff()
		end
	end
end

-- Получение точной роли игрока
local function GetRole(plr)
	if not isAlive(plr) then
		return "Dead"
	end

	if playerDataReady and playerData[plr.Name] then
		local data = playerData[plr.Name]
		if data.Dead or data.Killed then
			return "Dead"
		end
		
		if data.Role == "Murderer" then
			return "Murderer"
		elseif data.Role == "Sheriff" then
			return "Sheriff"
		elseif data.Role == "Hero" then
			return "Hero" -- Если ваш ESP красит только "Sheriff", замените эту строку на: return "Sheriff"
		elseif data.Role == "Innocent" then
			-- Резервная проверка: если сервер еще не обновил роль, но пистолет уже в руках
			if hasTool(plr, "Gun") then
				return "Hero"
			end
			return "Innocent"
		end
	end

	-- Fallback, если playerData еще не загрузилась
	if hasTool(plr, "Knife") then return "Murderer" end
	if hasTool(plr, "Gun") then return "Sheriff" end
	
	return "Dead"
end

local function setupPlayerData()
	local remotes = game.ReplicatedStorage:FindFirstChild("Remotes")
	if remotes then
		local gameplay = remotes:FindFirstChild("Gameplay")
		if gameplay then
			local playerDataChanged = gameplay:FindFirstChild("PlayerDataChanged")
			local GetCurrentPlayerData = gameplay:FindFirstChild("GetCurrentPlayerData")
			if playerDataChanged and GetCurrentPlayerData then
				local data = GetCurrentPlayerData:InvokeServer()
				playerData = data
				playerDataReady = true
				updateRoleCache()
				
				playerDataChanged.OnClientEvent:Connect(function(newData)
					playerData = newData
					playerDataReady = true
					-- Сбрасываем кэш при любом изменении данных раунда (смерть, подбор пистолета, новый раунд)
					cachedMurderer = nil
					cachedSheriff = nil
					updateRoleCache()
					if UpdatePlayerCheckboxes then UpdatePlayerCheckboxes() end
					if RefreshTargetInfo then RefreshTargetInfo() end
				end)
				return true
			end
		end
	end
	return false
end

local function GetColor(role)
	if role == "Murderer" then return Color3.fromRGB(255, 50, 50) end
	if role == "Sheriff"  then return Color3.fromRGB(50, 50, 255) end
	if role == "Innocent" then return Color3.fromRGB(50, 255, 50) end
	if role == "Hero" then return Color3.fromRGB(255, 255, 50) end
	return Color3.fromRGB(199, 199, 199)
end

local playerHighlights = {}

function RefreshHighlights()
	for _, hl in pairs(playerHighlights) do hl:Destroy() end
	playerHighlights = {}
	if not _G.ESP.EnabledPlayers() then return end

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr == LocalPlayer then continue end
		local char = plr.Character
		if not char then continue end
		local role = GetRole(plr)
		-- ✅ убрал "if role == Dead then continue end" — мёртвые снова видны (серым)
		local color = GetColor(role)
		local hl = Instance.new("Highlight")
		hl.Name = "PlayerHL"
		hl.FillTransparency = 0.5
		hl.OutlineTransparency = 0
		hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		hl.FillColor = color
		hl.OutlineColor = color
		hl.Adornee = char
		hl.Parent = ESPFolder
		hl.Enabled = true
		playerHighlights[plr] = hl
	end
end

function SetupPlayerTracking(player)
	if not player then return end

		RefreshAll()
	end

	player.CharacterAdded:Connect(onChanged)
	player.CharacterRemoving:Connect(onChanged)

	local function onToolAdded(tool)
		if tool:IsA("Tool") and (tool.Name == "Knife" or tool.Name == "Gun") then
			onChanged()
		end
	end

	if player.Character then
		player.Character.ChildAdded:Connect(onToolAdded)
	end
	player.CharacterAdded:Connect(function(char)
		char.ChildAdded:Connect(onToolAdded)
		local backpack = player:FindFirstChild("Backpack")
		if backpack then
			backpack.ChildAdded:Connect(onToolAdded)
		else
			player.ChildAdded:Connect(function(child)
				if child.Name == "Backpack" then
					child.ChildAdded:Connect(onToolAdded)
				end
			end)
		end
	end)
end

-- ============ ОЦЕНКА ДВИЖЕНИЯ ============

local function resetMotion(s)
	table.clear(s.hist)
	table.clear(s.queue)
	s.epoch += 1
	s.vel, s.acc, s.turn = nil, nil, nil
	s.stableHead, s.lastRev, s.wasMoving = nil, nil, false
	s.predPos, s.predDir = nil, nil
end

local function pushSample(s, now, pos)
	local h = s.hist
	local last = h[#h]
	if last and (pos - last.pos).Magnitude > TELEPORT_DIST then
		resetMotion(s) -- телепорт или респавн
	end
	table.insert(h, { t = now, pos = pos })
	while #h > 2 and now - h[1].t > HISTORY_MAX do
		table.remove(h, 1)
	end
end

-- Делим окно истории пополам: скорость на первой и второй половине даёт и скорость, и ускорение.
-- Возвращает vel, acc, turn или nil, если данных мало.
local function estimateMotion(h, window)
	local n = #h
	if n < 4 then return nil end

	local last = h[n]
	local startIdx = n
	for i = 1, n do
		if last.t - h[i].t <= window then
			startIdx = i
			break
		end
	end
	if n - startIdx + 1 < 4 then return nil end

	local first = h[startIdx]
	local span = last.t - first.t
	if span < window * 0.6 then return nil end

	local midT = first.t + span / 2
	local mid = first
	for i = startIdx + 1, n do
		if h[i].t >= midT then
			mid = h[i]
			break
		end
	end

	local d1, d2 = mid.t - first.t, last.t - mid.t
	if d1 < 0.03 or d2 < 0.03 then return nil end

	local v1 = (mid.pos - first.pos) / d1
	local v2 = (last.pos - mid.pos) / d2
	local dtc = (d1 + d2) / 2

	local acc = (v2 - v1) / dtc
	if acc.Magnitude > MAX_ACCEL then
		acc = acc.Unit * MAX_ACCEL
	end
	local vel = v2 + acc * (d2 / 2) -- переносим скорость на текущий момент

	local f1 = Vector3.new(v1.X, 0, v1.Z)
	local f2 = Vector3.new(v2.X, 0, v2.Z)
	local turn = 0
	if f1.Magnitude > MIN_SPEED and f2.Magnitude > MIN_SPEED then
		turn = math.clamp(wrapAngle(headingOf(f2) - headingOf(f1)) / dtc, -MAX_TURN, MAX_TURN)
	end

	return vel, acc, turn
end

local function updateMotion(s)
	local vel, acc, turn = estimateMotion(s.hist, active.window)
	if vel then
		s.vel, s.acc, s.turn = vel, acc, turn
	end
end

-- ============ АВТОПОДБОР ПАРАМЕТРОВ ============
-- Каждые SAMPLE_INTERVAL делаем пробные предсказания всеми вариантами параметров (упрощённая
-- модель движения по земле, без лучей). Через T секунд сравниваем с тем, где персонаж оказался
-- на самом деле, и копим среднюю ошибку каждого варианта. Лучший становится активным.
-- Горизонт T равен текущему времени предсказания, поэтому выбор сам подстраивается под пинг.

local function groundDisplacement(vel, acc, turn, accW, turnW, T, walkSpeed)
	local flat = Vector3.new(vel.X, 0, vel.Z)
	local flatAcc = Vector3.new(acc.X, 0, acc.Z) * accW
	local tr = turn * turnW
	local cap = math.max(walkSpeed, flat.Magnitude)
	local disp = Vector3.zero
	local elapsed = 0
	while elapsed < T do
		local dt = math.min(SIM_STEP, T - elapsed)
		elapsed += dt
		if elapsed <= ACC_TIME then
			flat += flatAcc * dt
		end
		if tr ~= 0 and flat.Magnitude > MIN_SPEED then
			local ang = tr * dt
			local c, sn = math.cos(ang), math.sin(ang)
			flat = Vector3.new(flat.X * c + flat.Z * sn, 0, -flat.X * sn + flat.Z * c)
		end
		if flat.Magnitude > cap then
			flat = flat.Unit * cap
		end
		disp += flat * dt
	end
	return disp
end

local function isGroundedNormal(hum)
	if hum.Sit or hum.FloorMaterial == Enum.Material.Air then
		return false
	end
	local st = hum:GetState()
	return st ~= Enum.HumanoidStateType.Climbing and st ~= Enum.HumanoidStateType.Swimming
end

-- ============ ПАТТЕРНЫ ДВИЖЕНИЯ И ОБУЧЕНИЕ ============
-- Профиль игрока: средняя скорость, развороты (стрейф A-D), остановки.
-- Поверх базового предсказания работает обучаемая поправка (онлайн-регрессия) в системе
-- координат "вдоль движения / вбок". Она применяется, только если на уже виденных данных
-- снижает ошибку относительно базового предсказания, иначе не используется.

local function updatePattern(s, now, dt)
	local v = s.vel
	if not v then return end
	local flat = Vector3.new(v.X, 0, v.Z)
	local sp = flat.Magnitude
	if sp >= PATTERN_MOVE_SPEED then
		s.wasMoving = true
		s.avgSpeed = s.avgSpeed and (s.avgSpeed + (sp - s.avgSpeed) * (1 - math.exp(-0.5 * dt))) or sp
		local h = headingOf(flat)
		if not s.stableHead then
			s.stableHead = h
		elseif math.abs(wrapAngle(h - s.stableHead)) > REVERSE_ANGLE then
			-- разворот: запоминаем интервал между разворотами (период стрейфа)
			if s.lastRev then
				local interval = now - s.lastRev
				if interval < 3 then
					s.meanRev += (math.clamp(interval, 0.1, 2) - s.meanRev) * 0.2
				end
			end
			s.lastRev = now
			s.revCount += 1
			s.stableHead = h
		else
			s.stableHead += wrapAngle(h - s.stableHead) * (1 - math.exp(-6 * dt))
		end
	elseif sp < MIN_SPEED and s.wasMoving then
		s.wasMoving = false
		s.stopCount += 1
	end
end

local function dotw(w, x)
	local r = 0
	for i = 1, N_FEAT do
		r += w[i] * x[i]
	end
	return r
end

-- Признаки в системе координат движения: скорость, ускорение вдоль и вбок, поворот,
-- фаза стрейфа (сколько прошло с последнего разворота относительно обычного периода)
local function buildFeatures(s, walk, now)
	local v = s.vel
	if not v then return nil end
	local flat = Vector3.new(v.X, 0, v.Z)
	local sp = flat.Magnitude
	if sp < MIN_SPEED then return nil end

	local fwd = flat / sp
	local side = Vector3.new(fwd.Z, 0, -fwd.X)
	local a = s.acc or Vector3.zero
	local sr = sp / math.max(walk, 1)
	local phase = 0
	if s.revCount >= 3 and s.lastRev then
		phase = math.clamp((now - s.lastRev) / s.meanRev, 0, 2) - 1
	end

	local x = {
		1,
		sr,
		a:Dot(fwd) / MAX_ACCEL,
		a:Dot(side) / MAX_ACCEL,
		(s.turn or 0) / MAX_TURN,
		phase,
		phase * sr,
	}
	return x, fwd, side
end

-- Один шаг обучения: сравниваем базовое предсказание и предсказание с поправкой (до обновления
-- весов, то есть честная проверка на новых данных), затем подстраиваем веса.
local function learnUpdate(m, L, actual)
	local T = L.T
	if T < 0.02 then return end
	local x = L.x

	local cf, cs = dotw(m.wf, x), dotw(m.ws, x)
	local corrected = L.base + (L.fwd * cf + L.side * cs) * T
	local eB = math.min((actual - L.base).Magnitude, MAX_ERR)
	local eC = math.min((actual - corrected).Magnitude, MAX_ERR)

	m.n += 1
	local r = math.max(LEARN_SCORE_RATE, 1 / m.n)
	local ob, oc = m.eBase or eB, m.eCorr or eC
	m.eBase = ob + (eB - ob) * r
	m.eCorr = oc + (eC - oc) * r

	-- целевая поправка к скорости, studs/сек, в системе "вдоль / вбок"
	local resid = actual - L.base
	local ef = resid:Dot(L.fwd) / T - cf
	local es = resid:Dot(L.side) / T - cs
	local xx = 1e-3
	for i = 1, N_FEAT do
		xx += x[i] * x[i]
	end
	for i = 1, N_FEAT do
		m.wf[i] = math.clamp(m.wf[i] + LEARN_MU * ef * x[i] / xx, -LEARN_MAX_W, LEARN_MAX_W)
		m.ws[i] = math.clamp(m.ws[i] + LEARN_MU * es * x[i] / xx, -LEARN_MAX_W, LEARN_MAX_W)
	end
end

local function modelUsable(m)
	return m.n >= LEARN_MIN and m.eBase ~= nil and m.eCorr ~= nil and m.eCorr < m.eBase * LEARN_GAIN_MARGIN
end

-- Поправка к предсказанной позиции (Vector3) или nil, если обученной модели, которой можно верить, нет
local function learnedCorrection(s, hum, now, T)
	if not LEARN or not s.vel or not isGroundedNormal(hum) then return nil end
	local x, fwd, side = buildFeatures(s, hum.WalkSpeed, now)
	if not x then return nil end

	local best = nil
	if modelUsable(s.model) then
		best = s.model
	end
	if modelUsable(globalModel) and (not best or globalModel.eCorr < best.eCorr) then
		best = globalModel
	end
	if not best then return nil end

	local corr = (fwd * dotw(best.wf, x) + side * dotw(best.ws, x)) * T
	local speed = Vector3.new(s.vel.X, 0, s.vel.Z).Magnitude
	local lim = MAX_CORR_RATIO * speed * T
	if corr.Magnitude > lim and corr.Magnitude > 0 then
		corr = corr.Unit * lim
	end
	return corr
end

local function logShadow(s, hum, upper, now, T)
	if s.nextSample and now < s.nextSample then return end
	s.nextSample = now + SAMPLE_INTERVAL
	if not isGroundedNormal(hum) then return end

	local est = {}
	for wi, w in ipairs(WINDOWS) do
		local v, a, tr = estimateMotion(s.hist, w)
		if not v then return end
		est[wi] = { v = v, a = a, tr = tr }
	end
	local mid = est[math.ceil(#WINDOWS / 2)].v
	if Vector3.new(mid.X, 0, mid.Z).Magnitude < MIN_SPEED then return end -- стоит: замер ничего не даст

	local walk = hum.WalkSpeed
	local preds = table.create(#candidates)
	for i, c in ipairs(candidates) do
		local e = est[c.wi]
		preds[i] = groundDisplacement(e.v, e.a, e.tr, c.accW, c.turnW, T, walk)
	end

	-- данные для обучения поправки: признаки и базовое предсказание активной модели
	local learn = nil
	if LEARN and s.vel then
		local x, fwd, side = buildFeatures(s, walk, now)
		if x then
			local base = groundDisplacement(s.vel, s.acc or Vector3.zero, s.turn or 0,
				active.accW, active.turnW, T, walk)
			learn = { x = x, fwd = fwd, side = side, base = base, T = T }
		end
	end

	table.insert(s.queue, {
		due = now + T,
		start = upper.Position,
		preds = preds,
		learn = learn,
		epoch = s.epoch,
	})
end

local function scoreQueue(s, hum, upper, now)
	local q = s.queue
	while q[1] and now >= q[1].due do
		local e = table.remove(q, 1)
		-- не считаем замер, если был телепорт, прыжок, падение или лестница
		if e.epoch == s.epoch and isGroundedNormal(hum) then
			local p = upper.Position
			local actual = Vector3.new(p.X - e.start.X, 0, p.Z - e.start.Z)
			scoredCount += 1
			local rate = math.max(SCORE_RATE, 1 / scoredCount)
			for i, d in ipairs(e.preds) do
				local err = math.min((actual - d).Magnitude, MAX_ERR)
				local old = scores[i] or err
				scores[i] = old + (err - old) * rate
			end
			if e.learn then
				learnUpdate(s.model, e.learn, actual)
				learnUpdate(globalModel, e.learn, actual)
			end
		end
	end
end

local function selectBest(now)
	if not AUTOTUNE then return end
	if now - lastSelect < SELECT_INTERVAL or scoredCount < MIN_SCORED then return end
	lastSelect = now

	local bestI, bestScore = nil, math.huge
	for i = 1, #candidates do
		local sc = scores[i]
		if sc and sc < bestScore then
			bestI, bestScore = i, sc
		end
	end
	if not bestI then return end
	if activeIndex and scores[activeIndex] and bestScore >= scores[activeIndex] * SWITCH_MARGIN then
		return
	end

	activeIndex = bestI
	local c = candidates[bestI]
	active = { window = c.window, accW = c.accW, turnW = c.turnW }
	if DEBUG then
		print(string.format("[Predict] window=%.2f acc=%.1f turn=%.1f  ошибка=%.2f studs (замеров: %d)",
			c.window, c.accW, c.turnW, bestScore, scoredCount))
	end
end

-- Для отладки: _G.GetPredictStats() -> активные параметры, средняя ошибка (studs), число замеров
_G.GetPredictStats = function()
	return active, activeIndex and scores[activeIndex] or nil, scoredCount
end

-- Возвращает предсказанную позицию торса и горизонтальную скорость в конце предсказания
local function simulate(s, hum, upper, T, params)
	local pos = upper.Position
	local vel = s.vel or Vector3.zero
	local flat = Vector3.new(vel.X, 0, vel.Z)
	local vy = vel.Y
	local acc = s.acc or Vector3.zero
	local flatAcc = Vector3.new(acc.X, 0, acc.Z) * active.accW
	local turn = (s.turn or 0) * active.turnW
	local gravity = workspace.Gravity
	local grounded = hum.FloorMaterial ~= Enum.Material.Air

	-- Лестницы, вода, сиденья: просто линейно
	local state = hum:GetState()
	if hum.Sit or state == Enum.HumanoidStateType.Climbing or state == Enum.HumanoidStateType.Swimming then
		return pos + vel * T, flat
	end

	-- Высота торса над полом: запоминаем, пока стоит на земле
	if grounded then
		local groundHit = workspace:Raycast(pos, Vector3.new(0, -20, 0), params)
		if groundHit then
			s.offset = pos.Y - groundHit.Position.Y
		end
	end
	local offset = s.offset or (hum.HipHeight + 1)

	-- Прыжок: стоит на земле, но вертикальная скорость большая
	if grounded and vy > JUMP_VEL then
		grounded = false
	elseif grounded then
		vy = 0
	end

	local speedCap = math.max(hum.WalkSpeed, flat.Magnitude)

	local elapsed = 0
	while elapsed < T do
		local dt = math.min(SIM_STEP, T - elapsed)
		elapsed += dt

		-- Управление работает только на земле
		if grounded then
			if elapsed <= ACC_TIME then
				flat += flatAcc * dt
			end
			if turn ~= 0 and flat.Magnitude > MIN_SPEED then
				local ang = turn * dt
				local c, sn = math.cos(ang), math.sin(ang)
				flat = Vector3.new(flat.X * c + flat.Z * sn, 0, -flat.X * sn + flat.Z * c)
			end
			if flat.Magnitude > speedCap then
				flat = flat.Unit * speedCap
			end
		end

		-- Горизонталь с проверкой стен
		local step = flat * dt
		local dist = step.Magnitude
		if dist > 1e-4 then
			local dir = step / dist
			local hit = workspace:Raycast(pos, dir * (dist + BODY_RADIUS), params)
			if hit and math.abs(hit.Normal.Y) < 0.7 then
				local allowed = math.max(0, hit.Distance - BODY_RADIUS)
				pos += dir * math.min(dist, allowed)
				local n = Vector3.new(hit.Normal.X, 0, hit.Normal.Z)
				if n.Magnitude > 1e-3 then
					n = n.Unit
					local into = flat:Dot(n)
					if into < 0 then
						flat -= n * into -- скользим вдоль стены
					end
				end
			else
				pos += step
			end
		end

		-- Вертикаль
		if grounded then
			local hit = workspace:Raycast(pos, Vector3.new(0, -(offset + STEP_DOWN), 0), params)
			if hit then
				pos = Vector3.new(pos.X, hit.Position.Y + offset, pos.Z)
				vy = 0
			else
				grounded = false -- сошёл с края
				vy = 0
			end
		else
			vy -= gravity * dt
			local dy = vy * dt
			if vy <= 0 then
				local hit = workspace:Raycast(pos, Vector3.new(0, -(-dy + offset + 0.1), 0), params)
				if hit then
					pos = Vector3.new(pos.X, hit.Position.Y + offset, pos.Z)
					vy = 0
					grounded = true
				else
					pos += Vector3.new(0, dy, 0)
				end
			else
				local hit = workspace:Raycast(pos, Vector3.new(0, dy + 1.5, 0), params)
				if hit then
					vy = 0 -- упёрся головой в потолок
				else
					pos += Vector3.new(0, dy, 0)
				end
			end
		end
	end

	return pos, flat
end

local ping = nil
local lastDebug = 0

RunService.RenderStepped:Connect(function(dt)
	local now = os.clock()

	-- Пинг подхватывается автоматически, сглаживаем только скачки
	local rawPing = getPing()
	ping = ping and (ping + (rawPing - ping) * (1 - math.exp(-PING_SMOOTH * dt))) or rawPing
	local T = math.clamp(ping * PING_FACTOR + EXTRA_DELAY, 0, MAX_PREDICT)

	local alpha = 1 - math.exp(-SMOOTHING * dt)
	local dirAlpha = 1 - math.exp(-DIR_SMOOTH * dt)
	local params = buildRayParams()

	selectBest(now)

	if DEBUG and now - lastDebug > 5 then
		lastDebug = now
		print(string.format("[Predict] обучение: замеров=%d  ошибка базы=%.2f  с поправкой=%.2f studs",
			globalModel.n, globalModel.eBase or 0, globalModel.eCorr or 0))
	end

	for _, player in ipairs(Players:GetPlayers()) do
		if player == Players.LocalPlayer and not INCLUDE_LOCAL then
			removeMarker(player)
			continue
		end

		local s = getState(player)
		local char = player.Character
		local upper = char and char:FindFirstChild("UpperTorso")
		local hum = char and char:FindFirstChildOfClass("Humanoid")

		if not (upper and hum) then
			resetMotion(s)
			removeMarker(player)
			continue
		end

		pushSample(s, now, upper.Position)

		if hum.Health <= 0 then
			s.predPos, s.predDir = nil, nil
			removeMarker(player)
			continue
		end

		scoreQueue(s, hum, upper, now)
		updateMotion(s)
		updatePattern(s, now, dt)
		logShadow(s, hum, upper, now, T)

		local target, flatVel = simulate(s, hum, upper, T, params)

		-- Обученная поправка по паттернам движения этого игрока (если она реально снижает ошибку)
		local corr = learnedCorrection(s, hum, now, T)
		if corr then
			target += corr
		end

		-- Направление маркера: по скорости, а если стоит, то по взгляду торса
		local dir
		if flatVel.Magnitude > MIN_SPEED then
			dir = flatVel.Unit
		else
			local look = upper.CFrame.LookVector
			local flatLook = Vector3.new(look.X, 0, look.Z)
			if flatLook.Magnitude > 0.01 then
				dir = flatLook.Unit
			end
		end
		if dir then
			s.dir = s.dir and s.dir:Lerp(dir, dirAlpha).Unit or dir
		end
		local facing = s.dir or Vector3.new(0, 0, -1)

		-- Результат доступен через getPredictedPosition (работает и при выключенном маркере)
		s.predPos, s.predDir = target, facing

		if not TOGGLE_MARKER then
			removeMarker(player)
			continue
		end

		local marker = markers[player]
		if not marker then
			marker = createMarker(player)
			markers[player] = marker
		end

		local goal = CFrame.lookAt(target, target + facing)
		if (marker.Position - target).Magnitude > 50 then
			marker.CFrame = goal
		else
			marker.CFrame = marker.CFrame:Lerp(goal, alpha)
		end
	end
end)

-- Avaible Space

local DIRECTION_COUNT = 128   -- сколько направлений проверять по сфере (больше = точнее, но тяжелее)
local MIN_RADIUS = 1          -- минимальный радиус при откате, studs
local RADIUS_STEP = 0.1         -- на сколько уменьшать радиус, studs
local STAND_HEIGHT = 3        -- высота HumanoidRootPart над полом, studs
local WALL_OFFSET = 1.5       -- отступ от стены, studs
local GROUND_PROBE = 6        -- с какой высоты пускать луч вниз, studs

-- Направления равномерно распределены по сфере (Fibonacci sphere)
local DIRECTIONS = {}
local function buildDirections(count)
	local dirs = {}
	local golden = math.pi * (3 - math.sqrt(5))
	for i = 0, count - 1 do
		local y = 1 - (i / math.max(count - 1, 1)) * 2
		local r = math.sqrt(1 - y * y)
		local theta = golden * i
		table.insert(dirs, Vector3.new(math.cos(theta) * r, y, math.sin(theta) * r))
	end
	return dirs
end
DIRECTIONS = buildDirections(DIRECTION_COUNT)

local function getAllCharacters()
	local list = {}
	for _, p in ipairs(Players:GetPlayers()) do
		if p.Character then
			table.insert(list, p.Character)
		end
	end
	return list
end

local function buildParams(ignore)
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = ignore
	return params
end

-- Нет стенок на прямой между двумя точками
local function hasLineOfSight(from, to, params)
	return Workspace:Raycast(from, to - from, params) == nil
end

-- Ищет пол под точкой и возвращает позицию для стояния или nil
local function findGround(point, params)
	local start = point + Vector3.new(0, GROUND_PROBE, 0)
	local hit = Workspace:Raycast(start, Vector3.new(0, -(GROUND_PROBE + 20), 0), params)
	if not hit then return nil end
	return hit.Position + Vector3.new(0, STAND_HEIGHT, 0)
end

-- Кандидат в направлении dir на расстоянии r. Если по пути стена, ставим перед ней.
local function getCandidate(origin, dir, r, params)
	local result = Workspace:Raycast(origin, dir * r, params)
	if not result then
		return origin + dir * r
	end
	local dist = (result.Position - origin).Magnitude - WALL_OFFSET
	if dist < MIN_RADIUS then
		return nil
	end
	return origin + dir * dist
end

-- origin: Vector3, от какой точки ищем место
-- radius: максимальный радиус поиска, studs
-- ignore: необязательная таблица объектов, которые не блокируют луч
-- Возвращает Vector3. Если подходящего места нет, возвращает origin (0 studs).
function getAvailableSpace(origin: Vector3, radius: number, ignore: {Instance}?): Vector3
	local params = buildParams(ignore or getAllCharacters())

	local r = radius
	while r >= MIN_RADIUS do
		for _, dir in ipairs(DIRECTIONS) do
			local candidate = getCandidate(origin, dir, r, params)
			if candidate then
				local ground = findGround(candidate, params)
				if ground
					and (ground - origin).Magnitude >= MIN_RADIUS
					and hasLineOfSight(origin, ground, params) then
					return ground
				end
			end
		end
		r -= RADIUS_STEP
	end

	-- Ни одно направление не подошло на всех радиусах: остаёмся на месте
	return origin
end

-- Удобная обёртка: место вокруг игрока по его HumanoidRootPart
function getAvailableSpaceFor(player: Player, radius: number): Vector3?
	local char = player.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then return nil end
	return getAvailableSpace(hrp.Position, radius, { char })
end

local function shootMurderer()
	local localplayer = game.Players.LocalPlayer

	if localplayer ~= findSheriff() then
		return
	end

	local murderer = findMurderer()
	if not murderer then
		return
	end

	local murdererTorso = murderer.Character.UpperTorso
	-- Достаём пистолет (если его нет в руках)
	local gun = localplayer.Character:FindFirstChild("Gun")
	if not gun then
		local backpack = localplayer:FindFirstChild("Backpack")
		if backpack then
			gun = backpack:FindFirstChild("Gun")
			if gun then
				local hum = localplayer.Character:FindFirstChild("Humanoid")
				if hum then
					hum:EquipTool(gun)
				end
			end
		end
		if not gun then
			return
		end
	end

	local rightHand = localplayer.Character:FindFirstChild("RightHand")
	if not rightHand then	
		return
	end

	-- Отправляем выстрел через RemoteEvent "Shoot"
	local shootEvent = gun:FindFirstChild("Shoot")

-- Speed Glitch
_G.Movement.SpeedGlitchSpeed = 5 -- Базовая скорость (укажи свою)

local SpeedGlitch = nil

-- Функция запуска движения в воздухе
local function startSpeedGlitch(character, humanoid)
	if SpeedGlitch or not _G.Movement.SpeedGlitchEnabled() then return end

	SpeedGlitch = RunService.Heartbeat:Connect(function(delta)
		if not (character and humanoid and humanoid.Parent) then
			return
		end

		if humanoid.MoveDirection.Magnitude > 0 then
			character:TranslateBy(humanoid.MoveDirection * _G.Movement.SpeedGlitchSpeed * delta * 10)
		end
	end)
end

-- Функция остановки
local function stopSpeedGlitch()
	if SpeedGlitch then
		SpeedGlitch:Disconnect()
		SpeedGlitch = nil
	end
end

-- Настройка отслеживания для конкретного персонажа
local function SpeedGlitchsetupCharacter(character)
	stopSpeedGlitch() -- Сбрасываем старый цикл, если он был активен
	
	local humanoid = character:WaitForChild("Humanoid")

	-- Включаем в воздухе, отключаем на земле
	humanoid.StateChanged:Connect(function(oldState, newState)
		if newState == Enum.HumanoidStateType.Freefall or newState == Enum.HumanoidStateType.Jumping then
			startSpeedGlitch(character, humanoid)
		elseif newState == Enum.HumanoidStateType.Landed or newState == Enum.HumanoidStateType.Running then
			stopSpeedGlitch()
		end
	end)

	-- Отключаем при смерти персонажа
	humanoid.Died:Connect(stopSpeedGlitch)
end

-- Запускаем для текущего персонажа (если он уже загружен)
if player.Character then
	SpeedGlitchsetupCharacter(player.Character)
end

-- Автоматически подключаем заново после каждого возрождения
player.CharacterAdded:Connect(SpeedGlitchsetupCharacter)


-- Fling

local defaultDestroyHeight = workspace.FallenPartsDestroyHeight
pcall(function()
	workspace.FallenPartsDestroyHeight = -50000
end)

local activeFlingRunning = false
local savedStartCFrame = nil

local function applyHorizontalFling(character, pushDir, power, spinPower)
	local horizVel = Vector3.new(pushDir.X * power, 0, pushDir.Z * power)
	for _, part in ipairs(character:GetDescendants()) do
		if part:IsA("BasePart") then
			part.AssemblyLinearVelocity = horizVel
			part.AssemblyAngularVelocity = spinPower > 0 and Vector3.new(0, spinPower, 0) or Vector3.zero
		end
	end
end

local function StopFling()
	activeFlingRunning = false

	local endChar = player.Character
	local endRoot = endChar and endChar:FindFirstChild("HumanoidRootPart")
	local endHumanoid = endChar and endChar:FindFirstChildOfClass("Humanoid")

	if camera and endHumanoid then
		camera.CameraSubject = endHumanoid
	end
	if endRoot and endRoot.Parent and savedStartCFrame then
		applyHorizontalFling(endChar, Vector3.zero, 0, 0)
		task.delay(0.05, function()
			if endRoot and endRoot.Parent and savedStartCFrame then
				endRoot.CFrame = savedStartCFrame
				savedStartCFrame = nil
			end
			if endHumanoid and endHumanoid.Parent then
				endHumanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
			end
			if endChar and endChar.Parent then
				applyHorizontalFling(endChar, Vector3.zero, 0, 0)
			end
		end)
	end
end

local function flingPlayer(targetPlayer)
	if activeFlingRunning then StopFling() return end
	if not targetPlayer then return end
	activeFlingRunning = true

	pcall(function()
		workspace.FallenPartsDestroyHeight = -50000
	end)

	local startChar = player.Character
	local startRoot = startChar and startChar:FindFirstChild("HumanoidRootPart")
	savedStartCFrame = startRoot and startRoot.CFrame

	local movel = 0.1
	local lastTargetSpeed = 0
	local lastLookVec = nil

	while activeFlingRunning do
		local dt = RunService.Heartbeat:Wait()
		if dt <= 0 then dt = 0.016 end

		local targetChar = targetPlayer and targetPlayer.Character
		local targetRoot = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
		local targetHumanoid = targetChar and targetChar:FindFirstChildOfClass("Humanoid")

		if not (targetRoot and targetHumanoid) or targetHumanoid.Health <= 0 then
			break
		end

		local targetVel = targetRoot.AssemblyLinearVelocity
		local targetSpeed = targetVel.Magnitude

		if targetSpeed > 850 or (targetSpeed - lastTargetSpeed) > 610 then
			break
		end
		lastTargetSpeed = targetSpeed

		if targetRoot.Position.Y <= (defaultDestroyHeight + 15) then
			break
		end

		if camera then
			camera.CameraSubject = targetHumanoid
		end

		local myChar = player.Character
		local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")

		if not (myChar and myChar.Parent and myRoot and myRoot.Parent) then
			continue
		end

		local _, currentYRot, _ = myRoot.CFrame:ToEulerAnglesYXZ()

		local lookVec = Vector3.new(targetRoot.CFrame.LookVector.X, 0, targetRoot.CFrame.LookVector.Z)
		if lookVec.Magnitude > 0 then lookVec = lookVec.Unit end

		local rightVec = Vector3.new(targetRoot.CFrame.RightVector.X, 0, targetRoot.CFrame.RightVector.Z)
		if rightVec.Magnitude > 0 then rightVec = rightVec.Unit end

		local turnSign = 0
		local turnDelta = 0
		local calculatedAngVel = 0

		if lastLookVec and lookVec.Magnitude > 0 then
			turnDelta = (lookVec - lastLookVec).Magnitude
			turnSign = lastLookVec:Cross(lookVec).Y
			calculatedAngVel = math.asin(math.clamp(turnSign, -1, 1)) / dt
		end
		lastLookVec = lookVec

		local physAngVel = targetRoot.AssemblyAngularVelocity.Y
		local angVelY = math.abs(physAngVel) > math.abs(calculatedAngVel) and physAngVel or calculatedAngVel

		local isTurningRight = (turnSign < -0.012) or (angVelY < -0.4)
		local isTurningLeft = (turnSign > 0.012) or (angVelY > 0.4)
		local isTurning = isTurningRight or isTurningLeft or (turnDelta > 0.015)

		-- Всегда заходим сбоку и выбиваем в сторону
		local offsetDir
		if isTurningRight then
			-- Цель поворачивается вправо -> стоим справа и выбиваем влево
			offsetDir = rightVec
		elseif isTurningLeft then
			-- Цель поворачивается влево -> стоим слева и выбиваем вправо
			offsetDir = -rightVec
		else
			-- По умолчанию (прямо или стоит) -> стоим с левого бока и выбиваем в сторону (вправо)
			offsetDir = -rightVec
		end

		-- Вектор удара направлен строго перпендикулярно телу цели в сторону
		local pushDir = -offsetDir

		local baseCenter
		if targetSpeed > 1 or isTurning then
			local myPing = 0.03
			pcall(function() myPing = player:GetNetworkPing() end)

			local basePredictionTime = (myPing * 1.5) + 0.015
			local clampedVel = targetSpeed > 100 and (targetVel.Unit * 100) or targetVel

			local effectiveVel = clampedVel
			if isTurning then
				local predictedTurnAngle = math.clamp(angVelY * basePredictionTime * 1.0, -1.0, 1.0)
				if targetSpeed > 2 then
					effectiveVel = (CFrame.Angles(0, predictedTurnAngle, 0) * clampedVel) + (offsetDir * 3)
				else
					effectiveVel = (CFrame.Angles(0, predictedTurnAngle, 0) * lookVec * 10) + (offsetDir * 3)
				end
			end

			local predictOffset = (effectiveVel * 2.8) * basePredictionTime
			local posNoPredict = targetRoot.Position - (predictOffset * 0.01)
			local posMaxPredict = targetRoot.Position + (predictOffset * 2.5)

			local now = os.clock()
			local horizontalSpeed = 6
			local alpha = (math.sin(now * horizontalSpeed * math.pi * 2) + 1) * 0.5
			baseCenter = posNoPredict:Lerp(posMaxPredict, alpha)
		else
			baseCenter = targetRoot.Position
		end

		local slamPos = Vector3.new(baseCenter.X, targetRoot.Position.Y, baseCenter.Z) + (offsetDir * 1.1)
		local targetCFrame = CFrame.new(slamPos) * CFrame.Angles(0, currentYRot, 0)

		myRoot.CFrame = targetCFrame
		applyHorizontalFling(myChar, pushDir, 90000, 9999)

		RunService.RenderStepped:Wait()
		if myChar and myChar.Parent and myRoot and myRoot.Parent then
			myRoot.CFrame = targetCFrame
			applyHorizontalFling(myChar, pushDir, 90000, 9999)
		end

		RunService.Stepped:Wait()
		if myChar and myChar.Parent and myRoot and myRoot.Parent then
			myRoot.CFrame = targetCFrame
			applyHorizontalFling(myChar, Vector3.new(1, 0, 0), movel, 0)
			movel = -movel
		end
	end
	StopFling()
end

_G.Visuals.TRCLOCALONLY = makeSetting(false, refreshAll())

TracerEvent.OnClientEvent:Connect(function(hnd, strp, endp)
	if not hnd or not endp or not _G.Visuals.TRCENABLED() then return end
	local startPos = hnd.Position
	local endPos = endp
	local plrt = hnd.Parent.Parent
	if _G.Visuals.TRCLOCALONLY() and plrt.Name ~= LocalPlayer.Name then return end
	-- 1. Рассчитываем пропорцию мягких краёв в зависимости от длины луча
	local distance = (endPos - startPos).Magnitude
	if distance < 0.1 then return end -- Защита от нулевой длины выстрела

	-- Ограничиваем до 0.49, чтобы начало и конец затухания не пересеклись на близкой дистанции
	local edgeRatio = math.clamp(_G.Visuals.TRCFADE_STUDS / distance, 0.001, 0.49)

	-- 2. Создаём точки привязки в мировых координатах
	local att0 = Instance.new("Attachment")
	att0.WorldPosition = startPos
	att0.Parent = workspace.Terrain

	local att1 = Instance.new("Attachment")
	att1.WorldPosition = endPos
	att1.Parent = workspace.Terrain

	-- 3. Создаём и настраиваем Beam
	local beam = Instance.new("Beam")
	beam.Attachment0 = att0
	beam.Attachment1 = att1
	beam.Width0 = 0.5
	beam.Width1 = 0.5
	beam.Texture = "rbxassetid://13797539713"
	beam.TextureMode = Enum.TextureMode.Stretch
	beam.LightEmission = 1
	beam.LightInfluence = 0
	beam.FaceCamera = true
	beam.Color = ColorSequence.new(Color3.fromHex("ff80e1"))
	beam.Transparency = NumberSequence.new({
		NumberSequenceKeypoint.new(0, 1),
		NumberSequenceKeypoint.new(edgeRatio, 0),
		NumberSequenceKeypoint.new(1 - edgeRatio, 0),
		NumberSequenceKeypoint.new(1, 1)
	})
	beam.Parent = workspace.Terrain

	-- 4. Плавное исчезновение с сохранением постоянной длины мягких краёв
	local fadeValue = Instance.new("NumberValue")
	fadeValue.Value = 0

	local connection = fadeValue.Changed:Connect(function(newAlpha)
		beam.Transparency = NumberSequence.new({
			NumberSequenceKeypoint.new(0, 1),
			NumberSequenceKeypoint.new(edgeRatio, newAlpha),
			NumberSequenceKeypoint.new(1 - edgeRatio, newAlpha),
			NumberSequenceKeypoint.new(1, 1)
		})
	end)

	local tweenInfo = TweenInfo.new(_G.Visuals.TRClifeTime, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
	local tween = TweenService:Create(fadeValue, tweenInfo, {Value = 1})
	tween:Play()

	-- 5. Безопасная очистка
	tween.Completed:Once(function()
		connection:Disconnect()
		fadeValue:Destroy()
		beam:Destroy()
		att0:Destroy()
		att1:Destroy()
	end)
end)

local function FormatPlayerText(Plr)
    local Role = GetRole(Plr)
    local RoleColor = GetColor(Role)
    local HexColor = "#FFFFFF" -- Цвет по умолчанию (белый)

    if typeof(RoleColor) == "Color3" then
        HexColor = "#" .. RoleColor:ToHex()
    elseif type(RoleColor) == "string" then
        -- Добавляем решетку, если GetColor вернул строку без нее (например, "FF0000")
        HexColor = RoleColor:sub(1, 1) == "#" and RoleColor or ("#" .. RoleColor)
    end

    return string.format('%s [<font color="%s">%s</font>]', Plr.DisplayName, HexColor, tostring(Role))
end

-- Включаем поддержку RichText внутри TextLabel чекбокса
local function EnableRichText(Checkbox)
    if Checkbox.TextLabel then
        Checkbox.TextLabel.RichText = true
    elseif Checkbox.Label then
        Checkbox.Label.RichText = true
    elseif Checkbox.Holder then
        local Label = Checkbox.Holder:FindFirstChildWhichIsA("TextLabel", true)
        if Label then
            Label.RichText = true
        end
    end
end

-- Функция обновления текста и ролей у ВСЕХ созданных чекбоксов
function UpdatePlayerCheckboxes()
    for Plr, Checkbox in pairs(PlayerCheckboxes) do
        if Plr and Plr.Parent == Players and Checkbox then
            EnableRichText(Checkbox)
            local NewText = FormatPlayerText(Plr)
            
            if Checkbox.SetText then
                Checkbox:SetText(NewText)
            elseif Checkbox.TextLabel then
                Checkbox.TextLabel.Text = NewText
            end
        end
    end
end

-- Подгонка высоты Tabbox
local function UpdateTabboxSize()
    task.wait()
    if TargetTrollTab.Resize then TargetTrollTab:Resize() end
    if TargetTabBox.Resize then TargetTabBox:Resize() end
end

-- Выбор первого доступного игрока
local function SelectFirstAvailablePlayer()
    for Plr, Checkbox in pairs(PlayerCheckboxes) do
        if Plr and Plr.Parent == Players then
            SelectedTarget = Plr
            IsUpdating = true
            Checkbox:SetValue(true)
            IsUpdating = false
            return
        end
    end
    SelectedTarget = nil
end

-- Создание чекбокса
local function AddPlayerCheckbox(Plr)
    if Plr == LocalPlayer or PlayerCheckboxes[Plr] then return end

    local FlagName = "TrollTarget_" .. Plr.Name

    local Checkbox = TargetTrollTab:AddCheckbox(FlagName, {
        Text = FormatPlayerText(Plr), -- Сразу ставим текст с RichText цветом
        Default = false,
        Callback = function(Value)
            if IsUpdating then return end

            if Value then
                SelectedTarget = Plr
                IsUpdating = true
                for OtherPlr, OtherBox in pairs(PlayerCheckboxes) do
                    if OtherPlr ~= Plr and OtherBox.Value then
                        OtherBox:SetValue(false)
                    end
                end
                IsUpdating = false
            else
                if SelectedTarget == Plr and Plr.Parent == Players then
                    task.spawn(function()
                        task.wait()
                        local CurrentBox = PlayerCheckboxes[Plr]
                        if SelectedTarget == Plr and CurrentBox and not CurrentBox.Value then
                            IsUpdating = true
                            CurrentBox:SetValue(true)
                            IsUpdating = false
                        end
                    end)
                end
            end
        end,
    })

    -- Активируем RichText для нового чекбокса
    EnableRichText(Checkbox)
    PlayerCheckboxes[Plr] = Checkbox

    if not SelectedTarget then
        SelectedTarget = Plr
        IsUpdating = true
        Checkbox:SetValue(true)
        IsUpdating = false
    end

    UpdateTabboxSize()
end

-- Удаление чекбокса при выходе игрока
local function RemovePlayerCheckbox(Plr)
    local Checkbox = PlayerCheckboxes[Plr]
    if not Checkbox then return end

    local WasSelected = (SelectedTarget == Plr)
    if WasSelected then
        SelectedTarget = nil
    end

    if Checkbox.Holder then
        Checkbox.Holder:Destroy()
    elseif Checkbox.Container then
        Checkbox.Container:Destroy()
    else
        Checkbox:SetVisible(false)
    end

    local FlagName = "TrollTarget_" .. Plr.Name
    if Library.Toggles then Library.Toggles[FlagName] = nil end
    if Library.Options then Library.Options[FlagName] = nil end

    PlayerCheckboxes[Plr] = nil

    if WasSelected then
        SelectFirstAvailablePlayer()
    end

    UpdateTabboxSize()
end

-- Заполняем текущих игроков
for _, Plr in ipairs(Players:GetPlayers()) do
    AddPlayerCheckbox(Plr)
end

if not SelectedTarget then
    SelectFirstAvailablePlayer()
end

    RoleLabel.TextLabel.RichText = true

local function UpdateTargetInfo(TargetPlr)
    if TargetPlr and TargetPlr.Parent == Players then
        DisplayNameLabel:SetText("Display Name: " .. TargetPlr.DisplayName)
        UsernameLabel:SetText("Username: @" .. TargetPlr.Name)
        UserIdLabel:SetText("User ID: " .. tostring(TargetPlr.UserId))

        -- Получаем роль и конвертируем цвет из GetColor в HEX для RichText
        local Role = GetRole(TargetPlr)
        local RoleColor = GetColor(Role)
        local HexColor = "#FFFFFF"

        if typeof(RoleColor) == "Color3" then
            HexColor = "#" .. RoleColor:ToHex()
        elseif type(RoleColor) == "string" then
            HexColor = RoleColor:sub(1, 1) == "#" and RoleColor or ("#" .. RoleColor)
        end

        RoleLabel:SetText(string.format('Role: <font color="%s">%s</font>', HexColor, tostring(Role)))

        -- Асинхронная загрузка аватарки
        task.spawn(function()
            local Success, Thumbnail = pcall(function()
                return Players:GetUserThumbnailAsync(
                    TargetPlr.UserId,
                    Enum.ThumbnailType.HeadShot,
                    Enum.ThumbnailSize.Size150x150
                )
            end)

            -- Проверяем, что цель не сменилась, пока грузилась картинка
            if Success and Thumbnail and SelectedTarget == TargetPlr then
                if AvatarImage.SetImage then
                    AvatarImage:SetImage(Thumbnail)
                elseif AvatarImage.ImageLabel then
                    AvatarImage.ImageLabel.Image = Thumbnail
                end
            end
        end)
    else
        DisplayNameLabel:SetText("Display Name: -")
        UsernameLabel:SetText("Username: -")
        UserIdLabel:SetText("User ID: -")
        RoleLabel:SetText("Role: -")

        if AvatarImage.SetImage then
            AvatarImage:SetImage("rbxassetid://0")
        elseif AvatarImage.ImageLabel then
            AvatarImage.ImageLabel.Image = "rbxassetid://0"
        end
    end
end

-- Функция для ручного или периодического обновления информации о текущей цели
function RefreshTargetInfo()
    UpdateTargetInfo(SelectedTarget)
end
