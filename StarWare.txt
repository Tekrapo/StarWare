local function destroyWeatherFX()
    for _, e in ipairs(V.Weather._emitters) do pcall(function() e:Destroy() end) end
    V.Weather._emitters = {}; safe_destroy(V.Weather._fxPart); V.Weather._fxPart = nil
end
local function makeWeatherFX(kind)
    destroyWeatherFX()
    local part = Instance.new("Part")
    part.Name="PulseWeatherFX"; part.Size=Vector3.new(200,1,200)
    part.Anchored=true; part.CanCollide=false; part.CanQuery=false; part.CanTouch=false
    part.CastShadow=false; part.Transparency=1; part.Parent=workspace
    V.Weather._fxPart = part
    local function emit(cfg)
        local pe = Instance.new("ParticleEmitter")
        pe.Texture = cfg.tex or "rbxasset://textures/particles/sparkles_main.dds"
        pe.Rate = cfg.rate or 200; pe.Lifetime = cfg.life; pe.Speed = cfg.speed
        pe.SpreadAngle = cfg.spread or Vector2.new(20,20)
        pe.Rotation = cfg.rot or NumberRange.new(0,360); pe.RotSpeed = cfg.rotspd or NumberRange.new(-30,30)
        pe.Size = cfg.size; pe.Transparency = cfg.trans; pe.Color = cfg.color
        pe.LightEmission = cfg.le or 0.8; pe.LightInfluence = 0
        pe.Acceleration = cfg.acc or Vector3.new(0,-10,0)
        pe.EmissionDirection = Enum.NormalId.Bottom
        pe.VelocityInheritance = 0; pe.Drag = cfg.drag or 0; pe.ZOffset = cfg.zoff or 2
        pe.Parent = part; table.insert(V.Weather._emitters, pe)
    end
    if kind=="Rain" then
        emit({rate=500, life=NumberRange.new(0.35,0.55), speed=NumberRange.new(120,180),
            spread=Vector2.new(0,0), rot=NumberRange.new(90,90), rotspd=NumberRange.new(0,0),
            size=NumberSequence.new({NumberSequenceKeypoint.new(0,0.18),NumberSequenceKeypoint.new(1,0.05)}),
            trans=NumberSequence.new({NumberSequenceKeypoint.new(0,0.35),NumberSequenceKeypoint.new(1,0.7)}),
            color=ColorSequence.new(Color3.fromRGB(180,210,255)), le=0.2, acc=Vector3.new(0,-200,0), drag=0, zoff=4})
    elseif kind=="Snow" then
        emit({rate=220, life=NumberRange.new(3,5), speed=NumberRange.new(4,8),
            spread=Vector2.new(35,35), rot=NumberRange.new(0,360), rotspd=NumberRange.new(-30,30),
            size=NumberSequence.new({NumberSequenceKeypoint.new(0,0.3),NumberSequenceKeypoint.new(0.5,0.55),NumberSequenceKeypoint.new(1,0.1)}),
            trans=NumberSequence.new({NumberSequenceKeypoint.new(0,0.05),NumberSequenceKeypoint.new(1,0.55)}),
            color=ColorSequence.new(Color3.fromRGB(255,255,255)), le=0.8, acc=Vector3.new(0,-3,0), drag=0.5, zoff=4})
    elseif kind=="Sakura" then
        emit({rate=90, life=NumberRange.new(3,6), speed=NumberRange.new(2,5),
            spread=Vector2.new(180,180), rot=NumberRange.new(0,360), rotspd=NumberRange.new(-60,60),
            size=NumberSequence.new({NumberSequenceKeypoint.new(0,0.4),NumberSequenceKeypoint.new(0.5,0.5),NumberSequenceKeypoint.new(1,0.1)}),

end
local function clearForeignAtm()
    for _, o in ipairs(L:GetChildren()) do
        if (o:IsA("Atmosphere") or o:IsA("Clouds") or o:IsA("Sky")) and tostring(o.Name):sub(1,5) ~= "Pulse" then
            o:SetAttribute("_pHid", o.Parent); o.Parent = nil
        end
    end
end
local function restoreForeignAtm()
    for _, o in ipairs(L:GetChildren()) do
        if o:GetAttribute("_pHid") ~= nil then o.Parent = o:GetAttribute("_pHid"); o:SetAttribute("_pHid", nil) end
    end
end
function V.Weather.set(mode)
    clearForeignAtm()
    for _, n in ipairs({"PulseAtmosphere","PulseClouds","PulseSky"}) do
        local o = L:FindFirstChild(n); if o then o:Destroy() end
    end
    local atm = Instance.new("Atmosphere"); atm.Name="PulseAtmosphere"
    local cl = Instance.new("Clouds"); cl.Name="PulseClouds"
    local sky = Instance.new("Sky"); sky.Name="PulseSky"
    if mode=="Fog" then
        atm.Density=0.65; atm.Color=Color3.fromRGB(180,180,200); atm.Decay=Color3.fromRGB(120,120,150); atm.Haze=3
        cl.Cover=0.9; cl.Density=0.7
        L.FogEnd=250; L.FogStart=0; L.FogColor=Color3.fromRGB(180,180,200); L.Brightness=1.2; L.ClockTime=15
        destroyWeatherFX()
    elseif mode=="Rain" then
        atm.Density=0.5; atm.Color=Color3.fromRGB(90,95,110); atm.Decay=Color3.fromRGB(50,50,70)
        cl.Cover=1; cl.Density=1; cl.Color=Color3.fromRGB(80,80,90)
        L.FogEnd=120; L.FogStart=0; L.FogColor=Color3.fromRGB(90,90,110); L.Brightness=0.7; L.ClockTime=14
        makeWeatherFX("Rain")
    elseif mode=="Snow" then
        atm.Density=0.3; atm.Color=Color3.fromRGB(230,235,245); atm.Decay=Color3.fromRGB(200,210,230)
        cl.Cover=0.8; cl.Density=0.4; cl.Color=Color3.fromRGB(240,240,255)
        L.FogEnd=400; L.FogColor=Color3.fromRGB(230,235,245); L.Brightness=2; L.ClockTime=12
        makeWeatherFX("Snow")
    elseif mode=="Sakura" then
        atm.Density=0.4; atm.Color=Color3.fromRGB(255,220,235); atm.Decay=Color3.fromRGB(255,183,210)
        cl.Cover=0.6; cl.Density=0.4; cl.Color=Color3.fromRGB(255,220,235)
        L.FogEnd=350; L.FogColor=Color3.fromRGB(255,220,235); L.Brightness=1.8; L.ClockTime=13
        makeWeatherFX("Sakura")
    elseif mode=="Night" then
        atm.Density=0.3; atm.Color=Color3.fromRGB(30,30,60); atm.Decay=Color3.fromRGB(10,10,30)
        cl.Cover=0.2; cl.Density=0.2

function V.Weather.off()
    for _, n in ipairs({"PulseAtmosphere","PulseClouds","PulseSky"}) do
        local o=L:FindFirstChild(n); if o then o:Destroy() end
    end
    destroyWeatherFX(); restoreForeignAtm()
    L.FogEnd=100000; L.Brightness=2; V.Weather.on=false
end

V.Wings = {on=false, scale=1, model=nil, feathers={}, color1=Color3.fromRGB(255,255,255), color2=Color3.fromRGB(255,230,160)}
local function clearWings() if V.Wings.model then pcall(function() V.Wings.model:Destroy() end) end; V.Wings.model=nil; V.Wings.feathers={} end
function V.Wings.set(v, scale, c1, c2)
    V.Wings.on = v and true or false
    V.Wings.scale = scale or V.Wings.scale
    if c1 then V.Wings.color1=c1 end; if c2 then V.Wings.color2=c2 end
    if not v then clearWings(); return end
    local char = LP.Character; if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart"); if not hrp then return end
    clearWings()
    local model = Instance.new("Model"); model.Name="PulseWings"
    local root = Instance.new("Part"); root.Size=Vector3.new(0.2,0.2,0.2); root.Transparency=1
    root.CanCollide=false; root.Massless=true; root.Anchored=false; root.Parent=model
    local rw = Instance.new("Weld"); rw.Part0=hrp; rw.Part1=root; rw.C0=CFrame.new(0,0.5,0.6); rw.Parent=root
    for side=-1,1,2 do for i=1,10 do
        local t=(i-1)/9; local len=(1.8+3.2*(1-t))*V.Wings.scale
        local f=Instance.new("Part"); f.Name="PulseFeather"
        f.Size=Vector3.new(0.35*V.Wings.scale, 0.12*V.Wings.scale, len)
        f.Color=V.Wings.color1:Lerp(V.Wings.color2, t*0.7)
        f.Material=Enum.Material.Neon; f.Transparency=0.08; f.Anchored=false
        f.CanCollide=false; f.CanQuery=false; f.CanTouch=false; f.Massless=true
        local m=Instance.new("SpecialMesh",f); m.MeshType=Enum.MeshType.Wedge
        local pl=Instance.new("PointLight",f); pl.Color=f.Color; pl.Brightness=1.2; pl.Range=6
        f.Parent=model
        local angY=math.rad(side*(10+t*65)); local angX=math.rad(-25+t*70); local angZ=math.rad(side*(-5-t*25))
        local w=Instance.new("Weld"); w.Part0=root; w.Part1=f
        local c0=CFrame.new(0,0,0)*CFrame.Angles(angX,angY,angZ)*CFrame.new(0,0,-len*0.5); w.C0=c0; w.Parent=f
        table.insert(V.Wings.feathers, {part=f, weld=w, baseC0=c0, baseCol=f.Color, phase=t, side=side})
    end end
    model.Parent=char; V.Wings.model=model
end
local function updateWings()
    if not V.Wings.on or not V.Wings.model or not V.Wings.model.Parent then return end
    local t=os.clock()
    for _,d in ipairs(V.Wings.feathers) do
        if d.weld and d.weld.Parent and d.part and d.part.Parent then
            local flap=math.sin(t*3+d.phase*2)*math.rad(12)*(1-d.phase*0.3)
            d.weld.C0 = d.baseC0 * CFrame.Angles(0, d.side*flap, 0)
            local pulse=0.5+0.5*math.sin(t*4+d.phase*6)
            d.part.Color=d.baseCol:Lerp(V.Wings.color1, pulse*0.5)
            d.part.Transparency=0.05+0.12*(1-pulse)
        end
    end
end

function V.Chams.apply(char, color, mode)
    if not char then return end
    local st=V.Chams._st[char] or {ch={}, seen={}}; V.Chams._st[char]=st
    for _,p in ipairs(char:GetDescendants()) do
        if p:IsA("BasePart") and p.Transparency<1 then
            local ok=(mode=="self" and isMe(p)) or (mode=="team" and not isMe(p))
            if ok then
                if not st.seen[p] then st.seen[p]=true; st.ch[#st.ch+1]={p, p.Material, p.Color} end
                if p.Material~=V.Chams.mat or p.Color~=color then pcall(function() p.Material=V.Chams.mat; p.Color=color end) end

local roleCacheMod, roleCache, roleCacheStamp = nil, {}, 0
local function refreshRoleCache()
    local now=os.clock(); if now-roleCacheStamp<0.4 and next(roleCache) then return end
    roleCacheStamp=now
    if not roleCacheMod then
        local ok,m=pcall(function() return require(Replicated:WaitForChild("Modules"):WaitForChild("CurrentRoundClient")) end)
        if ok and type(m)=="table" then roleCacheMod=m end
    end
    local d=roleCacheMod and roleCacheMod.PlayerData; if type(d)~="table" then return end
    local new={}
    for name,info in pairs(d) do if type(info)=="table" and info.Role then new[name]=info.Role end end
    roleCache=new
end
local function alive(p)
    local c=p.Character; if not c then return false end
    local h=c:FindFirstChildOfClass("Humanoid"); return h and h.Health>0
end
local function getRole(pl)
    local bp=pl:FindFirstChildOfClass("Backpack"); local ch=pl.Character
    local function chk(t)
        if not t then return nil end
        if t:FindFirstChild("Knife") then return "m" end
        if t:FindFirstChild("Gun") then return "s" end
        return nil
    end
    return chk(bp) or chk(ch)
end
local function cachedRole(pl)
    refreshRoleCache(); local c=roleCache[pl.Name]
    if c=="Murderer" then return "m" end
    if c=="Sheriff" or c=="Hero" then return "s" end
    return getRole(pl)
end

local function SA_rayBlocked(a, b, ignore)
    local list = {workspace.CurrentCamera}
    for _,x in ipairs(ignore or {}) do table.insert(list, x) end
    local p=RaycastParams.new(); p.FilterType=Enum.RaycastFilterType.Exclude
    for _=1,8 do
        p.FilterDescendantsInstances=list
        local hit=workspace:Raycast(a, b-a, p)
        if not hit then return false end
        local i=hit.Instance
        if i.Transparency>=0.9 or not i.CanCollide then table.insert(list, i) else return true end
    end
    return true
end
local SA_BODY = {"UpperTorso","Torso","LowerTorso","Head","HumanoidRootPart","RightUpperArm","LeftUpperArm","Right Arm","Left Arm"}
local function SA_wallCheck(origin, targetChar)
    if not SA.wallCheck then return true end
    if not origin or not targetChar then return false end
    local myChar=LP.Character
    local bodies={}
    for _,n in ipairs(SA_BODY) do
        local p=targetChar:FindFirstChild(n); if p and p:IsA("BasePart") then bodies[#bodies+1]=p end
    end
    if #bodies==0 then return false end
    local ig={targetChar}; if myChar and myChar~=targetChar then table.insert(ig, myChar) end
    local function clearFrom(o)
        if not o then return false end
        for _,p in ipairs(bodies) do if not SA_rayBlocked(o, p.Position, ig) then return true end end
        return false
    end
    if clearFrom(origin) then return true end
    if SA.wallCheckCamera then
        local cp=workspace.CurrentCamera and workspace.CurrentCamera.CFrame.Position
        if cp and (cp-origin).Magnitude>0.5 and clearFrom(cp) then return true end
    end
    return false
end
local function SA_getVel(root)
    local list=SA_hist[root]; if not list or #list<2 then return root.AssemblyLinearVelocity end
    local samples={}; local N=#list
    for i=2,N do
        local a,b=list[i-1], list[i]; local dt=b[1]-a[1]
        if dt>0.005 and dt<0.3 then samples[#samples+1]={vx=(b[2].X-a[2].X)/dt,vy=(b[2].Y-a[2].Y)/dt,vz=(b[2].Z-a[2].Z)/dt,w=(i/N)*(1/dt)} end
    end
    if #samples==0 then return root.AssemblyLinearVelocity end
    local vx,vy,vz,w=0,0,0,0
    for _,s in ipairs(samples) do vx=vx+s.vx*s.w; vy=vy+s.vy*s.w; vz=vz+s.vz*s.w; w=w+s.w end
    if w<=0 then return root.AssemblyLinearVelocity end
    local v=Vector3.new(vx/w, vy/w, vz/w); if v.Magnitude>200 then v=v.Unit*200 end; return v
end
local function SA_predict(h2, root, lead)
    if not SA.autoPred then local v=SA_getVel(root); return Vector3.new(v.X*lead, math.clamp(v.Y*lead,-3,3), v.Z*lead) end
    local vPhys=root.AssemblyLinearVelocity; local md=h2 and h2.MoveDirection or Vector3.zero
    if h2 and h2.FloorMaterial==Enum.Material.Air then
        local dy=vPhys.Y*lead - 0.5*workspace.Gravity*lead*lead
        return Vector3.new(vPhys.X*lead, dy, vPhys.Z*lead)
    end
    local vMove=md.Magnitude>0.1 and (md.Unit*(h2 and h2.WalkSpeed or 16)) or Vector3.zero
    local vBase=vMove.Magnitude>0.5 and vMove or SA_getVel(root)
    if md.Magnitude>0.1 then
        local right=root.CFrame.RightVector; local lat=math.abs(right:Dot(md.Unit)); lead=lead+lat*0.04
    end
    return Vector3.new(vBase.X*lead, math.clamp(vBase.Y*lead,-3,3), vBase.Z*lead)
end
local function findMuzzle()
    local myChar=LP.Character; if not myChar then return nil end
    for _,n in ipairs({"GunRaycastAttachment","GunMuzzleAttachment","MuzzleAttachment","Muzzle","GunMuzzle","BarrelAttachment","FirePointAttachment","FirePoint"}) do
        local a=myChar:FindFirstChild(n, true)
        if a and a:IsA("Attachment") then return a.WorldPosition end
    end
    local myRoot=myChar:FindFirstChild("HumanoidRootPart"); return myRoot and myRoot.Position
end
local function aimResolve()
    if not SA.enabled then return nil end
    if not target_p or not target_p.Character then pcall(refresh_target); if not target_p then return nil end end
    local c=target_char; if not c or not c.Parent then pcall(refresh_target); c=target_char end
    if not c then return nil end
    local root=c:FindFirstChild("HumanoidRootPart"); if not root then return nil end
    local h2=c:FindFirstChildOfClass("Humanoid")
    local lead=math.clamp(EC.rtt, 0.05, 0.45); if lead<0.09 then lead=0.09 end
    local off=SA_predict(h2, root, lead)
    local muzzle=findMuzzle(); if not muzzle then return nil end
    if SA.wallCheck then
        if not (SA_wallCheck(muzzle, c) or (SA.wallCheckCamera and SA_wallCheck((workspace.CurrentCamera or {}).CFrame.Position, c))) then return nil end
    end
    local order = SA.part=="Head" and {"Head","UpperTorso","Torso","HumanoidRootPart","LowerTorso"} or {"UpperTorso","Torso","LowerTorso","HumanoidRootPart","Head"}
    local myChar=LP.Character; rp.FilterDescendantsInstances={myChar, workspace.CurrentCamera}
    local candidates={}
    for _,name in ipairs(order) do
        local part=c:FindFirstChild(name)
        if part and part:IsA("BasePart") then
            local aimPos=part.Position+off
            local hit=workspace:Raycast(muzzle, aimPos-muzzle, rp)
            if not hit or hit.Instance:IsDescendantOf(c) then table.insert(candidates, {pos=aimPos, dist=(aimPos-muzzle).Magnitude}) end
        end
    end
    if #candidates>0 then table.sort(candidates, function(a,b) return a.dist<b.dist end); return candidates[1].pos end
    return (root.Position + off)
end
local WSH = {}
local function get_ws_fn()
    local ok,m=pcall(function() return require(Replicated:WaitForChild("ClientServices"):WaitForChild("WeaponService")) end)
    if ok and type(m)=="table" then return m end
    return WSH.m
end
local function install_hooks()
    local cur=get_ws_fn(); if not cur then return end
    if cur~=WSH.m then WSH.m=cur; WSH.hooked={} end
    if not WSH.hooked then WSH.hooked={} end
    pcall(function() setreadonly(cur, false) end)
    local aimCandidates = {
        "GetMouseTargetCFrame","GetTargetPosition","GetTargetCFrame","GetAimPosition",
        "GetShootCFrame","GetShootTarget","GetBulletOrigin","CalculateAim",
        "ComputeTarget","GetFiringVector","GetShotOrigin","GetGunFireData",
        "GetAimTarget","GetAimCFrame","GetTargetPos","GetShotPosition","GetShootPosition",
        "GetBulletTarget","GetFirePosition","GetFireTarget","GetGunTarget",
    }
    for _,name in ipairs(aimCandidates) do
        if type(cur[name])=="function" and not WSH.hooked[name] then
            local orig=cur[name]
            cur[name]=function(self, ...)
                local ok_o, orig_r = pcall(orig, self, ...)
                local pt = aimResolve()
                if pt then
                    if ok_o and typeof(orig_r)=="CFrame" then return CFrame.lookAt(orig_r.Position, pt) end
                    if ok_o and typeof(orig_r)=="Vector3" then return pt end
                    local cam = workspace.CurrentCamera
                    if cam then return CFrame.lookAt(cam.CFrame.Position, pt) end
                    return CFrame.new(pt)
                end
                return orig_r
            end
            WSH.hooked[name]=orig
        end
    end
end

    local center = cam.ViewportSize * 0.5
    local best_target = nil
    local best_dist = math.huge
    for _,plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and plr.Character then
            local h = plr.Character:FindFirstChildOfClass("Humanoid")
            if h and h.Health > 0 then
                if TRIG.team and cachedRole(plr) ~= "m" and cachedRole(LP) == "m" then
                    -- team check pass (same role)
                elseif TRIG.team and cachedRole(plr) == cachedRole(LP) then
                    -- skip same role
                else
                    for _,hb in ipairs(TRIG.hb) do
                        local part_name = TRIG_HB[hb]
                        local part = part_name and plr.Character:FindFirstChild(part_name)
                        if part and part:IsA("BasePart") then
                            local sp = cam:WorldToViewportPoint(part.Position)
                            if sp.Z > 0 then
                                local d2 = (Vector2.new(sp.X, sp.Y) - Vector2.new(center.X, center.Y)).Magnitude
                                if d2 <= fov and d2 < best_dist then
                                    best_dist = d2; best_target = {plr=plr, part=part}
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    if best_target then
        if math.random(1,100) <= TRIG.hitChance then
            local ws = get_ws_fn()
            if ws then
                for _,name in ipairs({"FireServer","Fire","Shoot","FireGun","FireWeapon","TriggerGun"}) do
                    if type(ws[name]) == "function" then pcall(function() ws[name](ws) end); break end
                end
            end
            TRIG._last = now
        end
    end
end

local function getKnifeTarget()
    local myChar=LP.Character; local myRoot=myChar and myChar:FindFirstChild("HumanoidRootPart"); if not myRoot then return nil end
    local cam=workspace.CurrentCamera; if not cam then return nil end
    local rp2=RaycastParams.new(); rp2.FilterType=Enum.RaycastFilterType.Exclude
    rp2.FilterDescendantsInstances={myChar, cam}
    local hit=workspace:Raycast(cam.CFrame.Position, cam.CFrame.LookVector*KS.reach, rp2)
    if hit then
        local hc=hit.Instance:FindFirstAncestorOfClass("Model"); local hp=hc and Players:GetPlayerFromCharacter(hc)
        if hp and hp~=LP then
            local c=hp.Character; local tp=c and (c:FindFirstChild("HumanoidRootPart") or c:FindFirstChild("Head"))
            if tp then return tp end
        end
    end
    local best,bd=nil,KS.reach
    for _,p in ipairs(Players:GetPlayers()) do
        if p~=LP and p.Character then
            local tp=p.Character:FindFirstChild("HumanoidRootPart") or p.Character:FindFirstChild("Head")

            if tp and h and h.Health>0 then
                local d=(tp.Position-myRoot.Position).Magnitude
                if d<bd then bd=d; best=tp end
            end
        end

local function hookKnife(tool)
    if knifeHooked[tool] then return end
    local ev=tool:FindFirstChild("Events"); if not ev then return end
    local touched=ev:FindFirstChild("HandleTouched"); if not touched then return end
    local origFS=touched.FireServer; if type(origFS)~="function" then return end
    knifeHooked[tool]={origTouched=origFS, touched=touched}
    touched.FireServer=function(self, part, ...)
        if KS.enabled then
            local aimPart=getKnifeTarget(); if aimPart then pcall(function() origFS(self, aimPart, ...) end); return end
        end
        return origFS(self, part, ...)
    end
end
local function unhookKnife(tool)
    local h=knifeHooked[tool]; if not h then return end
    pcall(function() h.touched.FireServer=h.origTouched end); knifeHooked[tool]=nil
end
local function tryHookKnife()
    local c=LP.Character; if not c then return end
    local knife=c:FindFirstChild("Knife"); if knife then hookKnife(knife) end
    for tool,_ in pairs(knifeHooked) do if not tool.Parent or tool.Parent~=c then unhookKnife(tool) end end
end
getgenv().KNIFE_AIM_RESOLVE = function()
    if not KS.enabled then return nil end
    local t=getKnifeTarget(); if not t then return nil end
    return CFrame.new(t.Position)
end

local function bt_build()
    bt_destroy()
    local char=LP.Character; if not char then return end
    local rhrp=char:FindFirstChild("HumanoidRootPart"); if not rhrp then return end
    local prev=char.Archivable; char.Archivable=true
    local ok,m=pcall(function() return char:Clone() end); char.Archivable=prev
    if not ok or not m then return end
    local rparts={}; for _,o in char:GetDescendants() do if o:IsA("BasePart") then rparts[#rparts+1]=o end end
    local ci=0
    for _,o in m:GetDescendants() do
        if o:IsA("Script") or o:IsA("LocalScript") or o:IsA("ModuleScript") or o:IsA("Decal") or o:IsA("ParticleEmitter") or o:IsA("Light") or o:IsA("Sound") then
            pcall(function() o:Destroy() end)
        elseif o:IsA("BasePart") then
            o.Anchored=true; o.CanCollide=false; o.CanQuery=false; o.CanTouch=false; o.Massless=true; o.CastShadow=false
            if o.Name=="HumanoidRootPart" then o.Transparency=1 else o.Material=Enum.Material.ForceField; o.Color=BT.color; o.Transparency=0.3 end
            ci=ci+1; if rparts[ci] then BT.cache[#BT.cache+1]={o, rparts[ci]} end
        end
    end
    local h=m:FindFirstChildOfClass("Humanoid"); if h then pcall(function() h:Destroy() end) end
    m.Name="\0"; m.Parent=workspace; BT.model=m
end
local function bt_update()
    if not BT.on then return end
    local char=LP.Character; local rhrp=char and char:FindFirstChild("HumanoidRootPart"); if not rhrp then return end
    if not BT.model or not BT.model.Parent then bt_build(); if not BT.model then return end end
    local now=os.clock(); local base_cf=rhrp.CFrame
    if BT.n<BT.cap then BT.n=BT.n+1 else BT.first=BT.first%BT.cap+1 end
    local idx=(BT.first+BT.n-2)%BT.cap+1; local slot=BT.hist[idx]; if not slot then slot={}; BT.hist[idx]=slot end
    slot[1]=now; slot[2]=base_cf
    if now-BT.lastping>=0.2 then
        BT.lastping=now
        local ok,val=pcall(function() return Stats.Network.ServerStatsItem["Data Ping"]:GetValue()/1000 end)
        BT.ping=math.clamp((ok and val) or 0.15, 0.05, 0.6)
    end
    local target=now-BT.ping; local cf=base_cf
    for k=BT.n,1,-1 do local s=BT.hist[(BT.first+k-2)%BT.cap+1]; if s and s[1] and s[1]<=target then cf=s[2]; break end end
    while BT.n>0 and BT.hist[BT.first] and BT.hist[BT.first][1]<now-1 do BT.first=BT.first%BT.cap+1; BT.n=BT.n-1 end
    local baseInv=rhrp.CFrame:Inverse()
    for i=1,#BT.cache do local cp,rpv=BT.cache[i][1], BT.cache[i][2]; if cp and cp.Parent and rpv and rpv.Parent then cp.CFrame=cf*(baseInv*rpv.CFrame) end end

                hrp.Velocity=Vector3.new(9e8,9e8*10,9e8); hrp.RotVelocity=Vector3.new(9e9,9e9,9e9) end)

            local h=c:FindFirstChildOfClass("Humanoid"); if h then h:ChangeState(Enum.HumanoidStateType.GettingUp) end

            if V.Chams.selfOn and LP.Character then V.Chams.apply(LP.Character, V.Chams.selfColor, "self") end
            if V.Chams.gunOn and LP.Character then V.Chams.applyWeapon(LP.Character) end
        end
        roleAcc = roleAcc + dt
        if roleAcc >= 0.05 then roleAcc=0; roleEspTick() end
        arrowAcc = arrowAcc + dt
        if arrowAcc >= 0.08 then arrowAcc=0; arrowTick() end
        knifeAcc = knifeAcc + dt
        if knifeAcc >= 0.5 then knifeAcc=0; pcall(tryHookKnife) end
        if V.MapFF.on then
            local n=0
            for _,p in ipairs(workspace:GetDescendants()) do
                if n>500 then break end
                if p:IsA("BasePart") and p.Transparency<1 then
                    local ch=p:FindFirstAncestorWhichIsA("Model")
                    if not ch or not Players:GetPlayerFromCharacter(ch) then
                        if V.MapFF._orig[p]==nil then V.MapFF._orig[p]={p.Material, p.Color} end
                        if p.Material~=Enum.Material.ForceField then p.Material=Enum.Material.ForceField; p.Color=Color3.fromRGB(120,200,255); n=n+1 end
                    end
                end
            end
        end
    end)
end)

    workspace.Gravity=196.2

    pcall(function() V.Weather.off() end)
    pcall(function() V.Fullbright.set(false) end)
    pcall(function() V.Wings.set(false) end)
    pcall(function() V.Chams.clearAll() end)
    pcall(function() V.MapFF.set(false) end)
    pcall(function() V.Crosshair.set(false) end)
    pcall(function() for plr,_ in pairs(V.RoleESP._guis) do destroyRoleGui(plr) end end)
    pcall(function() for plr,a in pairs(V.RoleESP._arrows) do a:Destroy() end end)
    pcall(function() if V.RoleESP._rootGui then V.RoleESP._rootGui:Destroy() end end)

    safe_disconnect(pulseHb); pulseHb=nil
    safe_disconnect(mainHb); mainHb=nil
    safe_disconnect(idleConn); idleConn=nil
    safe_disconnect(charConn); charConn=nil
    safe_disconnect(bt_conn); bt_conn=nil
    pcall(bt_destroy)

    if m and WSH.hooked then
        pcall(function() setreadonly(m, false) end)
        for name,orig in pairs(WSH.hooked) do pcall(function() m[name]=orig end) end
        WSH.hooked=nil; WSH.m=nil
    end
